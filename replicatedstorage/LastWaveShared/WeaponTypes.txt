export type AnimSet = {
	Equip: string, Idle: string, Walk: string, Run: string,
	Jump: string, Fall: string, Fire: string, Reload: string,
}
export type SoundSet = {
	Fire: string, Reload: string, Empty: string,
	MagOut: string, MagIn: string, Nob: string,
}
export type PartNames = { Root: string, Upper: string, Magazine: string, Trigger: string, Nob: string }

export type Mech = {
	UpperTravel: number,   -- ход затвора назад (студы)
	UpperReturn: number,   -- скорость возврата затвора
	TriggerAngle: number,  -- поворот курка (градусы)
	MagDropDist: number,   -- насколько магазин уезжает вниз
	MagDropTilt: number,   -- наклон магазина при выпадении
	NobTravel: number,     -- ход рукоятки взведения
	Timeline: {
		MagOut: number, MagGone: number, MagIn: number, MagSeated: number,
		NobPull: number, NobBack: number,
	},
	BoltHoldOpen: boolean, -- затвор остаётся открытым при пустом магазине
}

export type Recoil = {
	Kick: number, Rise: number, Roll: number,
	Stiffness: number, Damping: number,
	Camera: number, Sway: number,
}

export type WeaponDef = {
	Name: string, Slot: string,
	Damage: number, HeadshotMultiplier: number, FireRate: number,
	MagSize: number, Reserve: number, ReserveMax: number, ReloadTime: number,
	Range: number, Pellets: number, SpreadDeg: number,
	PropDamageMultiplier: number, Auto: boolean,
	AmmoPack: number, AmmoCost: { [string]: number },
	Model: string, Grip: CFrame,
	Parts: PartNames, Mech: Mech, Recoil: Recoil,
	Anim: AnimSet, Sound: SoundSet,
}

local function grip(x: number, y: number, z: number, rx: number, ry: number): CFrame
	return CFrame.new(x, y, z) * CFrame.Angles(math.rad(rx), math.rad(ry), 0)
end

local function parts(): PartNames
	return { Root = "Root", Upper = "Upper", Magazine = "Magazine", Trigger = "Trigger", Nob = "ReloadNob" }
end

local Tiers = {
	[1] = { Damage = 1.00, Mag = 1.00 },
	[2] = { Damage = 1.15, Mag = 1.20 },
	[3] = { Damage = 1.32, Mag = 1.40 },
}

local M = {}
M.Defs = Defs
M.Tiers = Tiers

function M.Get(id: string?): WeaponDef?
	if type(id) ~= "string" then return nil end
	return Defs[id]
end

function M.Tier(tier: number?)
	return Tiers[math.clamp(tier or 1, 1, 3)]
end

function M.MagSize(id: string, tier: number?): number
	local d = Defs[id]
	if not d then return 0 end
	return math.floor(d.MagSize * M.Tier(tier).Mag)
end

function M.SlotOf(id: string): string
	local d = Defs[id]
	return if d then d.Slot else "Primary"
end

function M.IsValidId(id: any): boolean
	return type(id) == "string" and id ~= "" and id ~= "rbxassetid://" and id ~= "rbxassetid://0"
end

return M
