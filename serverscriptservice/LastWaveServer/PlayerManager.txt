--!nonstrict
-- Команды, наборы, скорость (только сервер!), downed/спасение, превращение,
-- спавн зомби из ТВОИХ моделей и централизованный респавн с watchdog'ом.
local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local RunService = game:GetService("RunService")

local Shared = game.ReplicatedStorage.LastWaveShared
local Config = require(Shared.Config)
local Remotes = require(Shared.Remotes)
local Sanity = require(Shared.Sanity)
local Kits = require(Shared.SurvivorKits)
local ZombieTypes = require(Shared.ZombieTypes)

local AntiCheat = require(script.Parent.AntiCheat)
local PhysicsSetup = require(script.Parent.PhysicsSetup)
local EconomyManager = require(script.Parent.EconomyManager)
local ScoreManager = require(script.Parent.ScoreManager)
local WeaponManager = require(script.Parent.WeaponManager)
local MatchState = require(script.Parent.MatchState)
local Analytics = require(script.Parent.AnalyticsManager)
local ZombieRig = require(script.Parent.ZombieRig)

local PlayerManager = {}

local data = {}
local reviveJobs = {}
local spawnToken = {}   -- [player] = n, инвалидирует старые отложенные респавны
local pendingUntil = {} -- [player] = clock, когда сработает запланированный респавн
local lastSpawn = {}    -- [player] = clock последней загрузки тела

local function ensure(player: Player)
	data[player] = data[player] or {
		Team = "Lobby", Kit = "Builder", ZombieClass = "Walker",
		Downed = false, DownedUntil = 0, Sprinting = false,
		Stamina = Config.STAMINA_MAX, LastSprint = 0,
	}
	return data[player]
end

-- ── спавны: возвращаем ВСЕГДА валидный CFrame ─────────────────────────
local function collectSpawns(folderName: string, sector: string?)
	local folder = Workspace:FindFirstChild(folderName)
	if not folder then return {}, {} end
	local strict, loose = {}, {}
	for _, c in folder:GetDescendants() do
		if c:IsA("BasePart") then
			table.insert(loose, c)
			local sec = c:GetAttribute("Sector")
			if sector == nil or sec == nil or sec == sector then
				table.insert(strict, c)
			end
		end
	end
	return strict, loose
end

local function spawnCFrame(team: string): CFrame
	local folderName =
		if team == "Survivor" then "SurvivorSpawns"
		elseif team == "Zombie" then "ZombieSpawns"
		else "LobbySpawns"

	local strict, loose = collectSpawns(folderName, if team == "Zombie" then MatchState.SectorName() else nil)
	local list = if #strict > 0 then strict else loose

	if #list == 0 then
		-- Раньше здесь возвращался nil и тело падало в пустоту. Больше нет.
		warn(("[Spawn] Workspace.%s пустая или отсутствует. Ставлю на резервную точку."):format(folderName))
		local sl = Workspace:FindFirstChildWhichIsA("SpawnLocation", true)
		if sl then return sl.CFrame + Vector3.new(0, 5, 0) end
		local _, other = collectSpawns("LobbySpawns", nil)
		if #other > 0 then return other[1].CFrame + Vector3.new(0, 5, 0) end
		return CFrame.new(0, 25, 0)
	end

	if team == "Zombie" then
		-- не спавним в лицо выжившим, но если других точек нет - спавним хоть куда
		local survivors = PlayerManager.GetAllSurvivors()
		local safe = {}
		for _, part in list do
			local ok = true
			for _, p in survivors do
				local r = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
				if r and (r.Position - part.Position).Magnitude < 35 then ok = false; break end
			end
			if ok then table.insert(safe, part) end
		end
		if #safe > 0 then list = safe end
	end

	local chosen = list[math.random(1, #list)]
	return chosen.CFrame + Vector3.new(0, 3.5, 0)
end

-- ── загрузка тела ─────────────────────────────────────────────────────
local function loadZombieCharacter(player: Player, zClass: string): boolean
	if not Config.ZOMBIE_PLAYER_MODELS then return false end

	local model = ZombieRig.Build(zClass, false)
	if not model then return false end

	model.Name = player.Name
	local hum = model:FindFirstChildWhichIsA("Humanoid")
	if not hum then model:Destroy(); return false end
	hum.DisplayName = player.DisplayName

	model:PivotTo(spawnCFrame("Zombie"))

	local old = player.Character
	player.Character = nil
	if old then old:Destroy() end

	model.Parent = Workspace
	player.Character = model -- это и вызывает CharacterAdded -> OnCharacterSpawned
	ZombieRig.GiveOwnership(model, player)
	return true
end

-- Единственная точка спавна. Всё остальное только просит её сработать.
function PlayerManager.Spawn(player: Player, reason: string?)
	if not player.Parent then return end
	local d = ensure(player)

	if MatchState.Phase() == "GameOver" then
		pendingUntil[player] = nil
		return
	end

	-- защита от двойной загрузки: LoadCharacter в упор друг к другу = ошибка
	local now = os.clock()
	if now - (lastSpawn[player] or 0) < Config.RESPAWN_MIN_INTERVAL then
		PlayerManager.ScheduleRespawn(player, Config.RESPAWN_MIN_INTERVAL, "throttle")
		return
	end

	pendingUntil[player] = nil
	spawnToken[player] = (spawnToken[player] or 0) + 1
	lastSpawn[player] = now

	if d.Team == "Zombie" and loadZombieCharacter(player, d.ZombieClass or "Walker") then
		return
	end

	local ok, err = pcall(function() player:LoadCharacter() end)
	if not ok then
		warn(("[Spawn] LoadCharacter упал для %s: %s -> повтор через 1.5с"):format(player.Name, tostring(err)))
		PlayerManager.ScheduleRespawn(player, 1.5, "retry")
	end
end

function PlayerManager.ScheduleRespawn(player: Player, delay: number, reason: string?)
	if not player.Parent then return end
	spawnToken[player] = (spawnToken[player] or 0) + 1
	local myToken = spawnToken[player]
	pendingUntil[player] = os.clock() + delay

	task.delay(delay, function()
		if spawnToken[player] ~= myToken then return end -- перебит более свежим запросом
		PlayerManager.Spawn(player, reason)
	end)
end

-- ── команды ───────────────────────────────────────────────────────────
function PlayerManager.GetTeam(player: Player): string
	local d = data[player]
	return if d then d.Team else "Lobby"
end

function PlayerManager.SetKit(player: Player, kit: string)
	local d = ensure(player)
	if not Kits[kit] then return end
	d.Kit = kit
	if player.Character and d.Team == "Survivor" then
		player.Character:SetAttribute("Kit", kit)
	end
	Remotes.Notify:FireClient(player, "kitSet", kit)
end

function PlayerManager.SetZombieClass(player: Player, class: string)
	local d = ensure(player)
	local z = ZombieTypes[class]
	if not z then return end
	if z.UnlockWave > math.max(MatchState.Wave(), 1) then
		Remotes.Notify:FireClient(player, "classLocked", class)
		return
	end
	d.ZombieClass = class
	Remotes.Notify:FireClient(player, "classSet", class)
end

function PlayerManager.SetTeam(player: Player, team: string, zClass: string?)
	local d = ensure(player)
	d.Team = team
	d.Downed = false
	if zClass then d.ZombieClass = zClass end
	PlayerManager.Spawn(player, "setTeam")
end

function PlayerManager.OnCharacterSpawned(player: Player, char: Model)
	local d = ensure(player)
	local hum = char:WaitForChild("Humanoid", 5)
	local root = char:WaitForChild("HumanoidRootPart", 5)
	if not hum or not root then
		warn(("[Spawn] у %s нет Humanoid/HumanoidRootPart, повторяю спавн"):format(player.Name))
		PlayerManager.ScheduleRespawn(player, 1, "badRig")
		return
	end

	hum.BreakJointsOnDeath = false
	char:SetAttribute("Downed", false)
	char:SetAttribute("CarryMultiplier", 1)
	char:SetAttribute("Stamina", Config.STAMINA_MAX)
	char:SetAttribute("Sprinting", false)

	-- кастомную зомби-модель мы уже поставили на место, аватар ставим здесь
	if not char:GetAttribute("ZombieType") or d.Team ~= "Zombie" then
		char:PivotTo(spawnCFrame(d.Team))
	end

	if d.Team == "Survivor" then
		local kit = Kits[d.Kit] or Kits.Builder
		hum.MaxHealth = kit.MaxHealth
		hum.Health = kit.MaxHealth
		hum.WalkSpeed = kit.WalkSpeed
		hum.JumpPower = 50
		char:SetAttribute("Role", "Survivor")
		char:SetAttribute("Kit", d.Kit)
		char:SetAttribute("ZombieType", nil)

		if WeaponManager.GiveLoadout then
			WeaponManager.GiveLoadout(player, d.Kit)
		else
			WeaponManager.Give(player, kit.StartWeapon)
		end
		require(script.Parent.RigScale).ApplyPlayer(char, hum)
		require(script.Parent.HammerManager).Give(player)
		if next(kit.StartResources) then EconomyManager.Add(player, kit.StartResources) end

		task.defer(function()
			local ok, ShopManager = pcall(require, script.Parent.ShopManager)
			if ok and ShopManager.Refresh then ShopManager.Refresh(player) end
		end)

		Analytics.Funnel(player, 1, "spawned_survivor")

	elseif d.Team == "Zombie" then
		local z = ZombieTypes[d.ZombieClass] or ZombieTypes.Walker
		hum.MaxHealth = z.MaxHealth
		hum.Health = z.MaxHealth
		hum.WalkSpeed = z.WalkSpeed
		hum.JumpPower = z.JumpPower
		hum.RequiresNeck = false
		char:SetAttribute("Role", "Zombie")
		char:SetAttribute("ZombieType", z.Name)
		char:SetAttribute("AbilityReady", true)
		WeaponManager.Clear(player)
		require(script.Parent.RigScale).ApplyZombie(char, hum, z.Name)
		local bp = player:FindFirstChildOfClass("Backpack")
		local hammer = bp and bp:FindFirstChild("Hammer")
		if hammer then hammer:Destroy() end   -- зомби молотком не строит

		if char:GetAttribute("UsedFallbackRig") then
			warn(("[Spawn] %s заспавнен серой заглушкой: модель '%s' не найдена в ServerStorage.ZombieRigs"):format(player.Name, tostring(d.ZombieClass)))
		end
	else
		hum.WalkSpeed = 16
		hum.JumpPower = 50
		char:SetAttribute("Role", "Lobby")
		WeaponManager.Clear(player)
	end

	PhysicsSetup.ApplyToCharacter(char, char:GetAttribute("Role"))

	-- Смерть: один вход, две ветки, всегда через очередь
	local diedOnce = false
	local function onDied()
		if diedOnce then return end
		diedOnce = true
		if d.Team == "Survivor" then
			PlayerManager.ConvertToZombie(player)
		else
			PlayerManager.ScheduleRespawn(player, Config.ZOMBIE_RESPAWN_DELAY, "died")
		end
	end

	hum.Died:Connect(onDied)
	-- страховка: Died иногда не доезжает на кастомных ригах
	hum.HealthChanged:Connect(function(h)
		if h <= 0 then onDied() end
	end)
	char.AncestryChanged:Connect(function(_, parent)
		if parent == nil and player.Character == char and d.Team ~= "Lobby" then
			PlayerManager.ScheduleRespawn(player, Config.ZOMBIE_RESPAWN_DELAY, "removed")
		end
	end)
end

-- ── урон (единственная точка) ─────────────────────────────────────────
function PlayerManager.ApplyDamage(targetChar: Model, amount: number, source: Player?)
	if not Sanity.num(amount, 0, 100000) then return end
	local hum = targetChar:FindFirstChildWhichIsA("Humanoid")
	if not hum or hum.Health <= 0 then return end

	local player = Players:GetPlayerFromCharacter(targetChar)
	if not player then
		hum:TakeDamage(amount) -- AI-зомби
		return
	end

	local d = data[player]
	if not d then return end
	if d.Team ~= "Survivor" then hum:TakeDamage(amount); return end

	if d.Downed then
		d.DownedUntil -= 1.5
		return
	end
	if hum.Health - amount > 0 then
		hum:TakeDamage(amount)
		return
	end
	PlayerManager.SetDowned(player, source)
end

-- ── downed / спасение ─────────────────────────────────────────────────
function PlayerManager.SetDowned(player: Player, source: Player?)
	local d = data[player]
	if not d or d.Downed then return end
	local char = player.Character
	if not char then return end
	local hum = char:FindFirstChildWhichIsA("Humanoid")
	if not hum then return end

	d.Downed = true
	d.DownedUntil = os.clock() + Config.DOWNED_TIME
	char:SetAttribute("Downed", true)
	hum.Health = Config.DOWNED_HEALTH
	hum.WalkSpeed = Config.DOWNED_WALKSPEED
	hum.JumpPower = 0
	hum:SetStateEnabled(Enum.HumanoidStateType.Jumping, false)

	if source then ScoreManager.Award(source, "SurvivorDowned") end
	Remotes.PlayerDowned:FireAllClients(player, Config.DOWNED_TIME)

	task.spawn(function()
		while d.Downed and player.Parent do
			task.wait(0.25)
			if os.clock() >= d.DownedUntil then
				if source then ScoreManager.Award(source, "SurvivorKilled") end
				PlayerManager.ConvertToZombie(player)
				return
			end
		end
	end)
end

function PlayerManager.Revive(player: Player, medic: Player)
	local d = data[player]
	if not d or not d.Downed then return end
	local char = player.Character
	if not char then return end
	local hum = char:FindFirstChildWhichIsA("Humanoid")
	if not hum then return end

	local kit = Kits[d.Kit] or Kits.Builder
	d.Downed = false
	char:SetAttribute("Downed", false)
	hum.Health = Config.REVIVE_HEALTH
	hum.WalkSpeed = kit.WalkSpeed
	hum.JumpPower = 50
	hum:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)

	ScoreManager.Award(medic, "Revive")
	Remotes.PlayerRevived:FireAllClients(player, medic)
	Analytics.Custom(medic, "revive_success")
end

local function onRevive(player: Player, target: any)
	if not Sanity.inst(target, "Player") then return end
	local medicChar = player.Character
	if not medicChar or medicChar:GetAttribute("Role") ~= "Survivor" then return end
	if medicChar:GetAttribute("Downed") then return end

	local td = data[target]
	if not td or not td.Downed then return end
	if reviveJobs[player] then return end

	local medicKit = Kits[data[player] and data[player].Kit or "Builder"] or Kits.Builder
	if EconomyManager.Get(player, "Medkit") <= 0 then
		Remotes.Notify:FireClient(player, "noMedkit")
		return
	end

	local duration = Config.REVIVE_TIME / medicKit.ReviveSpeedMultiplier
	reviveJobs[player] = { Target = target, Progress = 0 }

	task.spawn(function()
		local elapsed = 0
		while elapsed < duration do
			task.wait(0.2)
			elapsed += 0.2
			local job = reviveJobs[player]
			if not job or job.Target ~= target then return end
			local a = medicChar:FindFirstChild("HumanoidRootPart")
			local b = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
			if not a or not b or not td.Downed or medicChar:GetAttribute("Downed") then
				reviveJobs[player] = nil
				Remotes.ReviveProgress:FireClient(player, 0)
				return
			end
			if (a.Position - b.Position).Magnitude > Config.REVIVE_MAX_DIST then
				reviveJobs[player] = nil
				Remotes.ReviveProgress:FireClient(player, 0)
				return
			end
			Remotes.ReviveProgress:FireClient(player, elapsed / duration)
		end
		reviveJobs[player] = nil
		if EconomyManager.Spend(player, { Medkit = 1 }) then
			PlayerManager.Revive(target, player)
		end
		Remotes.ReviveProgress:FireClient(player, 0)
	end)
end

-- ── превращение ───────────────────────────────────────────────────────
function PlayerManager.ConvertToZombie(player: Player, class: string?)
	local d = data[player]
	if not d then return end
	if d.Team == "Zombie" then
		-- уже зомби, просто вернуть в бой
		PlayerManager.ScheduleRespawn(player, Config.ZOMBIE_RESPAWN_DELAY, "reconvert")
		return
	end

	d.Team = "Zombie"
	d.Downed = false
	d.ZombieClass = class or d.ZombieClass or "Walker"

	local char = player.Character
	if char then char:SetAttribute("Downed", false) end
	WeaponManager.Clear(player)
	Remotes.PlayerInfected:FireAllClients(player, d.ZombieClass)

	local options = {}
	for name, z in ZombieTypes do
		if z.UnlockWave <= math.max(MatchState.Wave(), 1) then table.insert(options, name) end
	end
	Remotes.KitOptions:FireClient(player, "Zombie", options)
	Analytics.Funnel(player, 3, "became_zombie")

	PlayerManager.ScheduleRespawn(player, Config.INFECTION_RESPAWN_DELAY, "infected")
end

-- ── списки ────────────────────────────────────────────────────────────
function PlayerManager.GetAllSurvivors(): {Player}
	local out = {}
	for p, d in data do
		if d.Team == "Survivor" and p.Character then
			local hum = p.Character:FindFirstChildWhichIsA("Humanoid")
			if hum and hum.Health > 0 then table.insert(out, p) end
		end
	end
	return out
end

function PlayerManager.GetAliveStanding(): {Player}
	local out = {}
	for _, p in PlayerManager.GetAllSurvivors() do
		if not data[p].Downed then table.insert(out, p) end
	end
	return out
end

function PlayerManager.GetAllZombies(): {Player}
	local out = {}
	for p, d in data do
		if d.Team == "Zombie" then table.insert(out, p) end
	end
	return out
end

function PlayerManager.IsDowned(player: Player): boolean
	local d = data[player]
	return d ~= nil and d.Downed
end

-- ── стамина и скорость: ТОЛЬКО сервер ─────────────────────────────────
local function movementStep(dt: number)
	for player, d in data do
		local char = player.Character
		local hum = char and char:FindFirstChildWhichIsA("Humanoid")
		if not hum or hum.Health <= 0 then continue end

		if d.Downed then
			d.Sprinting = false
			char:SetAttribute("Sprinting", false)
		end

		if d.Team ~= "Survivor" then
			local z = ZombieTypes[d.ZombieClass] or ZombieTypes.Walker
			hum.WalkSpeed = if char:GetAttribute("Enraged") then z.WalkSpeed * 1.35 else z.WalkSpeed
			continue
		end

		if d.Downed then
			hum.WalkSpeed = Config.DOWNED_WALKSPEED
			continue
		end

		local kit = Kits[d.Kit] or Kits.Builder
		local carry = char:GetAttribute("CarryMultiplier") or 1
		local moving = hum.MoveDirection.Magnitude > 0.1

		if d.Sprinting and moving and d.Stamina > 0 then
			hum.WalkSpeed = kit.SprintSpeed * carry
			d.Stamina = math.max(0, d.Stamina - Config.STAMINA_DRAIN * dt)
			d.LastSprint = os.clock()
		else
			hum.WalkSpeed = kit.WalkSpeed * carry
			if os.clock() - d.LastSprint > Config.STAMINA_REGEN_DELAY then
				d.Stamina = math.min(Config.STAMINA_MAX, d.Stamina + Config.STAMINA_REGEN * dt)
			end
		end
		char:SetAttribute("Stamina", math.floor(d.Stamina))
	end
end

local function onSprint(player: Player, enabled: any)
	if type(enabled) ~= "boolean" then return end
	local d = data[player]
	if not d then return end
	local char = player.Character
	if not char then return end

	if d.Team ~= "Survivor" or d.Downed then
		d.Sprinting = false
		char:SetAttribute("Sprinting", false)
		return
	end
	d.Sprinting = enabled
	char:SetAttribute("Sprinting", enabled)
end

local function onUseMedkit(player: Player)
	local d = data[player]
	if not d or d.Team ~= "Survivor" or d.Downed then return end
	local char = player.Character
	local hum = char and char:FindFirstChildWhichIsA("Humanoid")
	if not hum or hum.Health >= hum.MaxHealth then return end
	if not EconomyManager.Spend(player, { Medkit = 1 }) then
		Remotes.Notify:FireClient(player, "noMedkit")
		return
	end
	hum.Health = math.min(hum.MaxHealth, hum.Health + 55)
end

-- ── watchdog: никто не остаётся трупом навсегда ───────────────────────
local function respawnWatchdog()
	if MatchState.Phase() == "GameOver" then return end
	local now = os.clock()
	for _, p in Players:GetPlayers() do
		local d = data[p]
		if not d then continue end
		if pendingUntil[p] and now < pendingUntil[p] + 2 then continue end

		local char = p.Character
		local hum = char and char:FindFirstChildWhichIsA("Humanoid")
		local bodyless = (not char) or (not char.Parent) or (not hum) or hum.Health <= 0
		if bodyless then
			warn(("[Respawn] %s застрял без тела (%s), форс-спавн"):format(p.Name, d.Team))
			PlayerManager.ScheduleRespawn(p, 0.25, "watchdog")
		end
	end
end

function PlayerManager.Init()
	local function hook(player: Player)
		ensure(player)
		player.CharacterAdded:Connect(function(char)
			PlayerManager.OnCharacterSpawned(player, char)
		end)
	end

	Players.PlayerAdded:Connect(hook)
	for _, p in Players:GetPlayers() do hook(p) end

	Players.PlayerRemoving:Connect(function(p)
		data[p] = nil
		reviveJobs[p] = nil
		spawnToken[p] = nil
		pendingUntil[p] = nil
		lastSpawn[p] = nil
	end)

	AntiCheat.Bind(Remotes.SprintRequest, "SprintRequest", onSprint)
	AntiCheat.Bind(Remotes.ReviveRequest, "ReviveRequest", onRevive)
	AntiCheat.Bind(Remotes.UseMedkitRequest, "UseMedkitRequest", onUseMedkit)

	AntiCheat.Bind(Remotes.KitSelectRequest, "KitSelectRequest", function(p, kit)
		if not Sanity.str(kit, 24) then return end
		local phase = MatchState.Phase()
		if phase ~= "Intermission" and phase ~= "Preparation" then return end
		PlayerManager.SetKit(p, kit)
	end)

	AntiCheat.Bind(Remotes.ZombieClassRequest, "ZombieClassRequest", function(p, class)
		if not Sanity.str(class, 24) then return end
		PlayerManager.SetZombieClass(p, class)
	end)

	RunService.Heartbeat:Connect(function(dt)
		local ok, err = pcall(movementStep, dt)
		if not ok then warn("[PlayerManager] movement: " .. tostring(err)) end
	end)

	task.spawn(function()
		while true do
			task.wait(Config.RESPAWN_WATCHDOG)
			local ok, err = pcall(respawnWatchdog)
			if not ok then warn("[PlayerManager] watchdog: " .. tostring(err)) end
		end
	end)
end

return PlayerManager
