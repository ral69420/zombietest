local Players=game:GetService("Players");local Workspace=game:GetService("Workspace");local RunService=game:GetService("RunService")
local Shared=game.ReplicatedStorage.LastWaveShared;local Config=require(Shared.Config);local Remotes=require(Shared.Remotes);local Sanity=require(Shared.Sanity);local Kits=require(Shared.SurvivorKits);local ZombieTypes=require(Shared.ZombieTypes)
local AntiCheat=require(script.Parent.AntiCheat);local Physics=require(script.Parent.PhysicsSetup);local Economy=require(script.Parent.EconomyManager);local Score=require(script.Parent.ScoreManager);local Weapons=require(script.Parent.WeaponManager);local Match=require(script.Parent.MatchState);local Analytics=require(script.Parent.AnalyticsManager);local ZombieRig=require(script.Parent.ZombieRig)
local M={};local data={};local reviveJobs={};local token={};local pending={};local lastSpawn={}
local function get(p)data[p]=data[p]or{Team="Lobby",Kit="Builder",ZombieClass="Walker",Downed=false,DownedUntil=0,Sprinting=false,Stamina=Config.STAMINA_MAX,LastSprint=0};return data[p]end
local function spawns(name,sector)local f=Workspace:FindFirstChild(name);local strict,loose={},{};if not f then return strict,loose end;for _,x in f:GetDescendants()do if x:IsA("BasePart")then table.insert(loose,x);local s=x:GetAttribute("Sector");if not sector or s==nil or s==sector then table.insert(strict,x)end end end;return strict,loose end
local function spawnCF(team)local name=team=="Survivor"and"SurvivorSpawns"or team=="Zombie"and"ZombieSpawns"or"LobbySpawns";local a,b=spawns(name,team=="Zombie"and Match.SectorName()or nil);local list=#a>0 and a or b;if #list==0 then local s=Workspace:FindFirstChildWhichIsA("SpawnLocation",true);return s and s.CFrame+Vector3.new(0,5,0)or CFrame.new(0,25,0)end;if team=="Zombie"then local safe={};for _,x in list do local ok=true;for _,p in M.GetAllSurvivors()do local r=p.Character and p.Character:FindFirstChild("HumanoidRootPart");if r and(r.Position-x.Position).Magnitude<35 then ok=false;break end end;if ok then table.insert(safe,x)end end;if #safe>0 then list=safe end end;return list[math.random(1,#list)].CFrame+Vector3.new(0,3.5,0)end
local function customZombie(p,class)local m=ZombieRig.Build(class,false);if not m then return false end;local h=m:FindFirstChildWhichIsA("Humanoid");if not h then m:Destroy();return false end;m.Name=p.Name;h.DisplayName=p.DisplayName;m:PivotTo(spawnCF("Zombie"));local old=p.Character;p.Character=nil;if old then old:Destroy()end;m.Parent=Workspace;p.Character=m;ZombieRig.GiveOwnership(m,p);return true end
function M.Spawn(p,reason)
 if not p.Parent then return end;local d=get(p);if Match.Phase()=="GameOver"then d.Team="Lobby";d.Downed=false end;local now=os.clock();if now-(lastSpawn[p]or 0)<Config.RESPAWN_MIN_INTERVAL then M.ScheduleRespawn(p,Config.RESPAWN_MIN_INTERVAL,"throttle");return end;pending[p]=nil;token[p]=(token[p]or 0)+1;lastSpawn[p]=now;if d.Team=="Zombie"and customZombie(p,d.ZombieClass or"Walker")then return end;local ok,e=pcall(function()p:LoadCharacter()end);if not ok then warn("[Spawn] "..tostring(e));M.ScheduleRespawn(p,1.5,"retry")end end
function M.ScheduleRespawn(p,delay,reason)if not p.Parent then return end;token[p]=(token[p]or 0)+1;local t=token[p];pending[p]=os.clock()+delay;task.delay(delay,function()if p.Parent and token[p]==t then M.Spawn(p,reason)end end)end
function M.GetTeam(p)return get(p).Team end
function M.SetKit(p,kit)if Kits[kit]then get(p).Kit=kit;if p.Character then p.Character:SetAttribute("Kit",kit)end;Remotes.Notify:FireClient(p,"kitSet",kit)end end
function M.SetZombieClass(p,class)local z=ZombieTypes[class];if not z then return end;if z.UnlockWave>math.max(Match.Wave(),1)then Remotes.Notify:FireClient(p,"classLocked",class);return end;get(p).ZombieClass=class;Remotes.Notify:FireClient(p,"classSet",class)end
function M.SetTeam(p,team,class)local d=get(p);d.Team=team;d.Downed=false;if class then d.ZombieClass=class end;M.Spawn(p,"setTeam")end
function M.OnCharacterSpawned(p,c)
 local d=get(p);local h=c:WaitForChild("Humanoid",5);local r=c:WaitForChild("HumanoidRootPart",5);if not h or not r then M.ScheduleRespawn(p,1,"badRig");return end;c:SetAttribute("Downed",false);c:SetAttribute("CarryMultiplier",1);c:SetAttribute("Stamina",Config.STAMINA_MAX);c:SetAttribute("Sprinting",false);if not c:GetAttribute("ZombieType")or d.Team~="Zombie"then c:PivotTo(spawnCF(d.Team))end
 if d.Team=="Survivor"then local k=Kits[d.Kit]or Kits.Builder;h.MaxHealth=k.MaxHealth;h.Health=k.MaxHealth;h.WalkSpeed=k.WalkSpeed;h.JumpPower=50;c:SetAttribute("Role","Survivor");c:SetAttribute("Kit",d.Kit);c:SetAttribute("ZombieType",nil);Weapons.GiveLoadout(p,d.Kit);require(script.Parent.RigScale).ApplyPlayer(c,h);require(script.Parent.HammerManager).Give(p);if next(k.StartResources)then Economy.Add(p,k.StartResources)end;Analytics.Funnel(p,1,"spawned_survivor")
 elseif d.Team=="Zombie"then local z=ZombieTypes[d.ZombieClass]or ZombieTypes.Walker;h.MaxHealth=z.MaxHealth;h.Health=z.MaxHealth;h.WalkSpeed=z.WalkSpeed;h.JumpPower=z.JumpPower;h.RequiresNeck=false;c:SetAttribute("Role","Zombie");c:SetAttribute("ZombieType",z.Name);c:SetAttribute("AbilityReady",true);Weapons.Clear(p);require(script.Parent.RigScale).ApplyZombie(c,h,z.Name);local b=p:FindFirstChildOfClass("Backpack");local hammer=b and b:FindFirstChild("Hammer");if hammer then hammer:Destroy()end
 else h.WalkSpeed=16;h.JumpPower=50;c:SetAttribute("Role","Lobby");Weapons.Clear(p)end
 Physics.ApplyToCharacter(c,c:GetAttribute("Role"));local once=false;local function died()if once then return end;once=true;if d.Team=="Survivor"then M.ConvertToZombie(p)else M.ScheduleRespawn(p,Config.ZOMBIE_RESPAWN_DELAY,"died")end end;h.Died:Connect(died);h.HealthChanged:Connect(function(v)if v<=0 then died()end end);c.AncestryChanged:Connect(function(_,parent)if not parent and p.Character==c and d.Team~="Lobby"then if d.Team=="Survivor"then M.ConvertToZombie(p)else M.ScheduleRespawn(p,Config.ZOMBIE_RESPAWN_DELAY,"removed")end end end)
end
function M.ApplyDamage(c,amount,source)if not Sanity.num(amount,0,100000)then return end;local h=c:FindFirstChildWhichIsA("Humanoid");if not h or h.Health<=0 then return end;local p=Players:GetPlayerFromCharacter(c);if not p then h:TakeDamage(amount);return end;local d=data[p];if not d or d.Team~="Survivor"then h:TakeDamage(amount);return end;if d.Downed then d.DownedUntil-=1.5;return end;if h.Health-amount>0 then h:TakeDamage(amount)else M.SetDowned(p,source)end end
function M.SetDowned(p,source)local d=data[p];local c=p.Character;local h=c and c:FindFirstChildWhichIsA("Humanoid");if not d or d.Downed or not h then return end;d.Downed=true;d.DownedUntil=os.clock()+Config.DOWNED_TIME;c:SetAttribute("Downed",true);h.Health=Config.DOWNED_HEALTH;h.WalkSpeed=Config.DOWNED_WALKSPEED;h.JumpPower=0;if source then Score.Award(source,"SurvivorDowned")end;Remotes.PlayerDowned:FireAllClients(p,Config.DOWNED_TIME);task.spawn(function()while d.Downed and p.Parent do task.wait(.25);if os.clock()>=d.DownedUntil then if source then Score.Award(source,"SurvivorKilled")end;M.ConvertToZombie(p);return end end end)end
function M.Revive(p,medic)local d=data[p];local c=p.Character;local h=c and c:FindFirstChildWhichIsA("Humanoid");if not d or not d.Downed or not h then return end;local k=Kits[d.Kit]or Kits.Builder;d.Downed=false;c:SetAttribute("Downed",false);h.Health=Config.REVIVE_HEALTH;h.WalkSpeed=k.WalkSpeed;h.JumpPower=50;Score.Award(medic,"Revive");Remotes.PlayerRevived:FireAllClients(p,medic)end
local function revive(p,target)if not Sanity.inst(target,"Player")then return end;local c=p.Character;local td=data[target];if not c or c:GetAttribute("Role")~="Survivor"or c:GetAttribute("Downed")or not td or not td.Downed or reviveJobs[p]then return end;if Economy.Get(p,"Medkit")<=0 then Remotes.Notify:FireClient(p,"noMedkit");return end;reviveJobs[p]={Target=target};local duration=Config.REVIVE_TIME/((Kits[data[p].Kit]or Kits.Builder).ReviveSpeedMultiplier);task.spawn(function()local elapsed=0;while elapsed<duration do task.wait(.2);elapsed+=.2;local j=reviveJobs[p];local a=c:FindFirstChild("HumanoidRootPart");local b=target.Character and target.Character:FindFirstChild("HumanoidRootPart");if not j or not a or not b or not td.Downed or c:GetAttribute("Downed")or(a.Position-b.Position).Magnitude>Config.REVIVE_MAX_DIST then reviveJobs[p]=nil;return end;Remotes.ReviveProgress:FireClient(p,elapsed/duration)end;reviveJobs[p]=nil;if Economy.Spend(p,{Medkit=1})then M.Revive(target,p)end;Remotes.ReviveProgress:FireClient(p,0)end)end
function M.ConvertToZombie(p,class)local d=data[p];if not d then return end;if d.Team=="Zombie"then M.ScheduleRespawn(p,Config.ZOMBIE_RESPAWN_DELAY,"reconvert");return end;d.Team="Zombie";d.Downed=false;d.ZombieClass=class or d.ZombieClass or"Walker";if p.Character then p.Character:SetAttribute("Downed",false)end;Weapons.Clear(p);Remotes.PlayerInfected:FireAllClients(p,d.ZombieClass);Analytics.Funnel(p,3,"became_zombie");M.ScheduleRespawn(p,Config.INFECTION_RESPAWN_DELAY,"infected")end
function M.GetAllSurvivors()local out={};for p,d in data do if d.Team=="Survivor"and p.Character then local h=p.Character:FindFirstChildWhichIsA("Humanoid");if h and h.Health>0 then table.insert(out,p)end end end;return out end
function M.GetAliveStanding()local out={};for _,p in M.GetAllSurvivors()do if not data[p].Downed then table.insert(out,p)end end;return out end
function M.GetAllZombies()local out={};for p,d in data do if d.Team=="Zombie"then table.insert(out,p)end end;return out end
function M.IsDowned(p)return data[p]~=nil and data[p].Downed end
local function movement(dt)for p,d in data do local c=p.Character;local h=c and c:FindFirstChildWhichIsA("Humanoid");if h and h.Health>0 then if d.Team~="Survivor"then local z=ZombieTypes[d.ZombieClass]or ZombieTypes.Walker;h.WalkSpeed=z.WalkSpeed;continue end;if d.Downed then h.WalkSpeed=Config.DOWNED_WALKSPEED;continue end;local k=Kits[d.Kit]or Kits.Builder;local moving=h.MoveDirection.Magnitude>.1;local carry=c:GetAttribute("CarryMultiplier")or 1;if d.Sprinting and moving and d.Stamina>0 then h.WalkSpeed=k.SprintSpeed*carry;d.Stamina=math.max(0,d.Stamina-Config.STAMINA_DRAIN*dt);d.LastSprint=os.clock()else h.WalkSpeed=k.WalkSpeed*carry;if os.clock()-d.LastSprint>Config.STAMINA_REGEN_DELAY then d.Stamina=math.min(Config.STAMINA_MAX,d.Stamina+Config.STAMINA_REGEN*dt)end end;c:SetAttribute("Stamina",math.floor(d.Stamina))end end end
function M.Init()local function hook(p)get(p);p.CharacterAdded:Connect(function(c)M.OnCharacterSpawned(p,c)end)end;Players.PlayerAdded:Connect(hook);for _,p in Players:GetPlayers()do hook(p)end;Players.PlayerRemoving:Connect(function(p)data[p]=nil;reviveJobs[p]=nil;token[p]=nil;pending[p]=nil;lastSpawn[p]=nil end);AntiCheat.Bind(Remotes.SprintRequest,"SprintRequest",function(p,v)if type(v)=="boolean"and data[p]then data[p].Sprinting=v end end);AntiCheat.Bind(Remotes.ReviveRequest,"ReviveRequest",revive);AntiCheat.Bind(Remotes.KitSelectRequest,"KitSelectRequest",function(p,k)if Sanity.str(k,24)and(Match.Phase()=="Intermission"or Match.Phase()=="Preparation")then M.SetKit(p,k)end end);AntiCheat.Bind(Remotes.ZombieClassRequest,"ZombieClassRequest",function(p,k)if Sanity.str(k,24)then M.SetZombieClass(p,k)end end);RunService.Heartbeat:Connect(function(dt)pcall(movement,dt)end);task.spawn(function()while true do task.wait(Config.RESPAWN_WATCHDOG);for _,p in Players:GetPlayers()do local d=data[p];local c=p.Character;local h=c and c:FindFirstChildWhichIsA("Humanoid");if d and not pending[p]and((not c)or not c.Parent or not h or h.Health<=0)then M.ScheduleRespawn(p,.25,"watchdog")end end end end)end
return M
