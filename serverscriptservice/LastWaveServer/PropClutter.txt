--!nonstrict
-- Вещи, которые лежат НА пропе. Стул, а на нём коробки.
-- Физика и коллизии включены всегда: игрок может толкать предметы и баррикады.
-- Спавнятся строго на поверхности объекта через рэйкаст.

local Workspace = game:GetService("Workspace")
local RunService = game:GetService("RunService")
local CollectionService = game:GetService("CollectionService")
local Debris = game:GetService("Debris")

local Config = require(game.ReplicatedStorage.LastWaveShared.Config)
local C = Config.CLUTTER

local PropClutter = {}
local registered = {}   -- [prop] = { BasePart, ... }
local active = {}       -- [part] = expireAt
local activeCount = 0
local debrisFolder: Folder

local function isClutter(inst: Instance): boolean
	if CollectionService:HasTag(inst, C.TagName) then return true end
	if inst:GetAttribute("Clutter") == true then return true end
	local cur = inst.Parent
	while cur and cur ~= Workspace do
		if cur.Name == C.FolderName then return true end
		cur = cur.Parent
	end
	return false
end

local function settleOnSurface(part: BasePart, prop: Instance)
	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Include
	rp.FilterDescendantsInstances = { prop }
	local hit = Workspace:Raycast(part.Position + Vector3.new(0, 1.5, 0), Vector3.new(0, -5, 0), rp)
	if hit then
		part.CFrame = CFrame.new(hit.Position + Vector3.new(0, part.Size.Y * 0.5 + 0.02, 0)) * (part.CFrame - part.Position)
	end
end

-- Регистрируем при спавне пропа: включаем физику и коллизию сразу, чтобы можно было толкать.
function PropClutter.Register(prop: Instance)
	if not C.Enabled then return end
	local list = {}
	for _, d in prop:GetDescendants() do
		if d:IsA("BasePart") and isClutter(d) then
			settleOnSurface(d, prop)
			d.Anchored = false
			d.Massless = false
			d.CanCollide = true
			d.CanQuery = true
			d.CanTouch = true
			d.CastShadow = false
			d.CollisionGroup = Config.GROUP_DEBRIS
			d.CustomPhysicalProperties = PhysicalProperties.new(C.Density, 0.5, 0.35, 1, 1)
			table.insert(list, d)
		end
	end
	if #list > 0 then
		registered[prop] = list
		prop:SetAttribute("HasClutter", true)
	end
end

function PropClutter.Forget(prop: Instance)
	registered[prop] = nil
end

-- Взял проп -> отсоединяем оставшиеся связи и переносим в Debris
function PropClutter.Release(prop: Instance)
	if not C.Enabled then return end
	local list = registered[prop]
	if not list then return end
	registered[prop] = nil
	prop:SetAttribute("HasClutter", false)

	local now = os.clock()
	for _, part in list do
		if not part.Parent then continue end
		if activeCount >= C.MaxActive then
			part:Destroy()
			continue
		end

		for _, j in part:GetChildren() do
			if j:IsA("JointInstance") or j:IsA("WeldConstraint") or j:IsA("Constraint") then
				j:Destroy()
			end
		end
		for _, j in prop:GetDescendants() do
			if (j:IsA("WeldConstraint") or j:IsA("JointInstance"))
				and (j.Part0 == part or j.Part1 == part) then
				j:Destroy()
			end
		end

		local cf = part.CFrame
		part.Parent = debrisFolder
		part.CFrame = cf
		part.Anchored = false
		part.Massless = false
		part.CanCollide = true
		part.CanQuery = true
		part.CollisionGroup = Config.GROUP_DEBRIS
		part.CustomPhysicalProperties = PhysicalProperties.new(C.Density, 0.5, 0.35, 1, 1)
		part.AssemblyLinearVelocity = Vector3.new(
			(math.random() - 0.5) * C.DropImpulse,
			0.5,
			(math.random() - 0.5) * C.DropImpulse
		)
		part.AssemblyAngularVelocity = Vector3.new(
			math.random(-3, 3), math.random(-3, 3), math.random(-3, 3)
		)

		active[part] = now + C.Lifetime
		activeCount += 1
	end
end

function PropClutter.ClearAll()
	table.clear(registered)
	for part in active do
		if part.Parent then part:Destroy() end
	end
	table.clear(active)
	activeCount = 0
end

function PropClutter.Init()
	debrisFolder = Workspace:FindFirstChild("BarricadeDebris")
	if not debrisFolder then
		debrisFolder = Instance.new("Folder")
		debrisFolder.Name = "BarricadeDebris"
		debrisFolder.Parent = Workspace
	end

	-- 10 Гц вместо каждого кадра: анти-флинг и уборка.
	task.spawn(function()
		while true do
			task.wait(0.1)
			local now = os.clock()
			for part, expire in active do
				if not part.Parent then
					active[part] = nil
					activeCount -= 1
					continue
				end
				if now > expire or part.Position.Y < -150 then
					active[part] = nil
					activeCount -= 1
					part:Destroy()
					continue
				end
				local v = part.AssemblyLinearVelocity
				if v.Magnitude > C.MaxSpeed then
					part.AssemblyLinearVelocity = v.Unit * C.MaxSpeed
				end
			end
		end
	end)
end

return PropClutter
