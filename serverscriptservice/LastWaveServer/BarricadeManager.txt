--!nonstrict
local Players=game:GetService("Players")
local Workspace=game:GetService("Workspace")
local ServerStorage=game:GetService("ServerStorage")
local RunService=game:GetService("RunService")
local CollectionService=game:GetService("CollectionService")
local Shared=game.ReplicatedStorage.LastWaveShared
local Config=require(Shared.Config);local Remotes=require(Shared.Remotes);local Sanity=require(Shared.Sanity);local PropTypes=require(Shared.PropTypes);local Kits=require(Shared.SurvivorKits);local Placement=require(Shared.PlacementRules)
local AntiCheat=require(script.Parent.AntiCheat);local PhysicsSetup=require(script.Parent.PhysicsSetup);local EconomyManager=require(script.Parent.EconomyManager);local ScoreManager=require(script.Parent.ScoreManager);local MatchState=require(script.Parent.MatchState);local PropClutter=require(script.Parent.PropClutter)
local BarricadeManager={};local B=Config.BARRICADE;local H=Config.HAMMER
local propState={};local carryByPlayer={};local carrierByProp={};local repairCd={};local dismantling={};local placedByPlayer={};local placedTotal=0;local propsFolder
local BUILD_PHASES={Intermission=true,Preparation=true,Wave=true,Transition=true,Evacuation=true}
local function mainPart(p)if p:IsA("BasePart")then return p end;if p:IsA("Model")then return p.PrimaryPart or p:FindFirstChildWhichIsA("BasePart")end end
local function eachPart(p,fn)if p:IsA("BasePart")then fn(p)end;for _,x in p:GetDescendants()do if x:IsA("BasePart")then fn(x)end end end
local function setAnchored(p,on)eachPart(p,function(x)x.Anchored=on end)end
local function setCollide(p,on)eachPart(p,function(x)x.CanCollide=on end)end
local function stage(r)if r>.8 then return 1 elseif r>.6 then return 2 elseif r>.35 then return 3 elseif r>.12 then return 4 else return 5 end end
local function sync(p)local s=propState[p];if not s then return end;p:SetAttribute("HP",math.floor(s.HP));p:SetAttribute("MaxHP",math.floor(s.MaxHP));p:SetAttribute("BaseMaxHP",math.floor(s.BaseMaxHP));p:SetAttribute("IsNailed",s.Nailed);p:SetAttribute("Stage",stage(s.HP/math.max(s.MaxHP,1)));p:SetAttribute("Infected",s.Infected or false);p:SetAttribute("PlacedBy",s.PlacedBy)end
local function kitOf(p)local c=p.Character;return Kits[c and c:GetAttribute("Kit") or "Builder"] or Kits.Builder end
local function ready(p)local c=p.Character;if not c or c:GetAttribute("Role")~="Survivor" or c:GetAttribute("Downed")then return false end;local h=c:FindFirstChildWhichIsA("Humanoid");return h and h.Health>0 end
local function headOf(p)local c=p.Character;local h=c and(c:FindFirstChild("Head")or c:FindFirstChild("HumanoidRootPart"));return h and h:IsA("BasePart")and h or nil end
local function inReach(p,prop,dist)local r=AntiCheat.GetRoot(p);return r and Placement.ReachDistance(r.Position,prop)<=dist end
local function mark(p,prop)local c=p.Character;if not c then return end;local v=c:FindFirstChild("CarriedProp")or Instance.new("ObjectValue");v.Name="CarriedProp";v.Parent=c;v.Value=prop end
local function settle(p)
 local size=Placement.SizeOf(p);local cf=p:IsA("Model")and p:GetBoundingBox()or p.CFrame;local rp=RaycastParams.new();rp.FilterType=Enum.RaycastFilterType.Exclude;rp.FilterDescendantsInstances={p,Workspace:FindFirstChild("BarricadeDebris")};rp.RespectCanCollide=true;local hit=Workspace:Raycast(cf.Position+Vector3.new(0,2,0),Vector3.new(0,-200,0),rp);if not hit then return end;local yaw=math.atan2(-cf.LookVector.X,-cf.LookVector.Z);Placement.ApplyCenter(p,CFrame.new(hit.Position+Vector3.new(0,size.Y*.5+.02,0))*CFrame.Angles(0,yaw,0))
end
function BarricadeManager.RegisterProp(item)
 local part=mainPart(item);if not part then return end;local key=tostring(item:GetAttribute("PropType")or item.Name);local def=PropTypes[key]or PropTypes.WoodCrate;if not PropTypes[key]then key="WoodCrate"end;local size=item:IsA("Model")and select(2,item:GetBoundingBox())or part.Size;item:SetAttribute("PropSize",size);item:SetAttribute("PropType",key);propState[item]={HP=def.MaxHP,MaxHP=def.MaxHP,BaseMaxHP=def.MaxHP,Nailed=false,PropType=key,Infected=false,PlacedBy=0,Size=size};CollectionService:AddTag(item,"BarricadeProp");PhysicsSetup.SetPropGroup(item,false);PropClutter.Register(item);setAnchored(item,true);setCollide(item,true);settle(item);sync(item)
end
function BarricadeManager.RegisterSocket(s)if not s:IsA("BasePart")then return end;s.Anchored=true;s.CanCollide=false;s.Transparency=1 end
function BarricadeManager.GetFreeSockets()return {} end
local function release(p,keep)local c=carryByPlayer[p];if not c then return end;carryByPlayer[p]=nil;carrierByProp[c.Prop]=nil;if c.Prop.Parent then c.Prop:SetAttribute("CarriedBy",nil);if not keep then PhysicsSetup.SetPropGroup(c.Prop,false);setCollide(c.Prop,true)end end;local ch=p.Character;if ch then ch:SetAttribute("CarryMultiplier",1)end;mark(p,nil);return c end
function BarricadeManager.TakeIntoHands(p,prop)
 if carryByPlayer[p]then return false end;local s=propState[prop];if not s or s.Nailed or carrierByProp[prop]then return false end;local def=PropTypes[s.PropType];carryByPlayer[p]={Prop=prop,Def=def};carrierByProp[prop]=p;setAnchored(prop,true);setCollide(prop,false);PhysicsSetup.SetPropGroup(prop,true);prop:SetAttribute("CarriedBy",p.UserId);mark(p,prop);PropClutter.Release(prop);Remotes.ClutterFx:FireAllClients("Drop",(mainPart(prop)or prop).Position);local c=p.Character;c:SetAttribute("CarryMultiplier",math.clamp(def.CarrySpeedMultiplier+kitOf(p).CarryBonus,.35,1));Remotes.Notify:FireClient(p,"carry",s.PropType);return true
end
local function onPickup(p,prop)
 if not Sanity.inst(prop,"Instance")or not ready(p)or carryByPlayer[p]then return end
 local HammerManager=require(script.Parent.HammerManager)
 if not HammerManager.IsEquipped(p)then Remotes.Notify:FireClient(p,"needHammer");return end
 local s=propState[prop];if not s or not CollectionService:HasTag(prop,"BarricadeProp")or not prop:IsDescendantOf(propsFolder)then AntiCheat.Flag(p,"pickup: не баррикада ("..prop.Name..")",4);return end;if s.Nailed then Remotes.Notify:FireClient(p,"nailedHint");return end;if not inReach(p,prop,Config.AC.MaxInteractDistance)then return end;BarricadeManager.TakeIntoHands(p,prop)
end
local function onDrop(p)local c=release(p);if not c or not c.Prop.Parent then return end;local r=AntiCheat.GetRoot(p);if not r then return end;Placement.ApplyCenter(c.Prop,CFrame.new(r.Position+r.CFrame.LookVector*4));settle(c.Prop)end
local function limitFor(p)return B.LimitPerPlayer+kitOf(p).ExtraPropLimit end
local function onPlace(p,dir,yaw)
 if not Sanity.unitVec(dir)then AntiCheat.Flag(p,"place: мусорный вектор",3);return end;if type(yaw)~="number"or yaw~=yaw or math.abs(yaw)>100 then yaw=0 end;if not ready(p)or not BUILD_PHASES[MatchState.Phase()]then return end;local c=carryByPlayer[p];local h=headOf(p);if not c or not h then return end;if(placedByPlayer[p.UserId]or 0)>=limitFor(p)then Remotes.Notify:FireClient(p,"propLimit");return end;if placedTotal>=B.LimitTotal then Remotes.Notify:FireClient(p,"propLimitTotal");return end;local cf,reason=Placement.Solve({origin=h.Position,dir=dir,maxDist=B.MaxPlaceDistance,size=propState[c.Prop].Size,yaw=yaw,ignore={p.Character,c.Prop}});if not cf then Remotes.Notify:FireClient(p,"placeFail",reason);return end;if next(c.Def.PlaceCost)and not EconomyManager.Spend(p,c.Def.PlaceCost)then Remotes.Notify:FireClient(p,"noResources");return end;release(p,true);Placement.ApplyCenter(c.Prop,cf);setAnchored(c.Prop,true);setCollide(c.Prop,true);PhysicsSetup.SetPropGroup(c.Prop,false);local s=propState[c.Prop];s.Nailed=true;s.PlacedBy=p.UserId;placedByPlayer[p.UserId]=(placedByPlayer[p.UserId]or 0)+1;placedTotal+=1;sync(c.Prop);ScoreManager.Award(p,"PropPlaced");Remotes.PropFx:FireAllClients("Place",cf.Position,s.PropType);pcall(function()require(script.Parent.AnalyticsManager).Funnel(p,2,"first_barricade")end)
end
function BarricadeManager.Repair(p,prop)
 if not Sanity.inst(prop,"Instance")or not ready(p)then return false end;local s=propState[prop];if not s or not s.Nailed then return false end;if s.Infected then Remotes.Notify:FireClient(p,"infected");return false end;if s.HP>=s.MaxHP then Remotes.Notify:FireClient(p,"repairFull");return false end;local now=os.clock();if(repairCd[p]or 0)+H.RepairCooldown-Config.AC.CooldownSlack>now or not inReach(p,prop,H.Reach)then return false end;if not EconomyManager.Spend(p,H.RepairCost)then Remotes.Notify:FireClient(p,"noResources");return false end;repairCd[p]=now;local kit=kitOf(p);s.MaxHP=math.max(s.BaseMaxHP*H.MinMaxHPRatio,s.MaxHP-s.BaseMaxHP*H.DegradePerRepair);s.HP=math.clamp(s.HP+H.RepairAmount*kit.RepairMultiplier,0,s.MaxHP);sync(prop);ScoreManager.Award(p,"PropRepaired");local part=mainPart(prop);if part then Remotes.PropFx:FireAllClients("Repair",part.Position,s.HP/s.MaxHP)end;return true
end
function BarricadeManager.BeginDismantle(p,prop)
 if not Sanity.inst(prop,"Instance")or not ready(p)then return end;local s=propState[prop];if not s or not s.Nailed then return end;if H.OwnOnly and s.PlacedBy~=p.UserId then Remotes.Notify:FireClient(p,"notYours");return end;if not inReach(p,prop,H.Reach)then return end;dismantling[p]={Prop=prop,Until=os.clock()+H.DismantleTime};Remotes.ReviveProgress:FireClient(p,.01)
end
function BarricadeManager.CancelDismantle(p)if dismantling[p]then dismantling[p]=nil;Remotes.ReviveProgress:FireClient(p,0)end end
local function unnail(prop)local s=propState[prop];if not s or not s.Nailed then return end;s.Nailed=false;s.Infected=false;if s.PlacedBy~=0 then placedByPlayer[s.PlacedBy]=math.max(0,(placedByPlayer[s.PlacedBy]or 1)-1)end;s.PlacedBy=0;placedTotal=math.max(0,placedTotal-1);sync(prop)end
local function finish(p,prop)local s=propState[prop];if not s or not s.Nailed then return end;local def=PropTypes[s.PropType];unnail(prop);if next(def.PlaceCost)and H.DismantleRefund>0 then local back={};for k,v in def.PlaceCost do back[k]=math.max(1,math.floor(v*H.DismantleRefund))end;EconomyManager.Add(p,back)end;local part=mainPart(prop);if part then Remotes.PropFx:FireAllClients("Salvage",part.Position,s.PropType)end;if H.DismantleToHands and not carryByPlayer[p]then BarricadeManager.TakeIntoHands(p,prop)else settle(prop)end;Remotes.Notify:FireClient(p,"dismantled")end
function BarricadeManager.DamageProp(prop,damage,source)local s=propState[prop];if not s or not s.Nailed or not Sanity.num(damage,0,100000)then return end;s.HP=math.clamp(s.HP-damage,0,s.MaxHP);sync(prop);local part=mainPart(prop);if part then Remotes.PropFx:FireAllClients("Damage",part.Position,s.HP/s.MaxHP)end;if s.HP>0 then return end;if source then ScoreManager.Award(source,"PropDestroyed")end;unnail(prop);local def=PropTypes[s.PropType];if part then for _,p in Players:GetPlayers()do local r=p.Character and p.Character:FindFirstChild("HumanoidRootPart");if r and p.Character:GetAttribute("Role")=="Survivor"and(r.Position-part.Position).Magnitude<40 then local y={};for k,v in def.SalvageYield do y[k]=math.max(1,math.floor(v*Config.SALVAGE_RETURN_RATIO))end;EconomyManager.Add(p,y)end end;Remotes.PropFx:FireAllClients("Break",part.Position,s.PropType)end;setCollide(prop,false);task.delay(Config.PROP_DESPAWN_TIME,function()PropClutter.Forget(prop);if prop.Parent then prop:Destroy()end;propState[prop]=nil;carrierByProp[prop]=nil end)end
function BarricadeManager.InfectProp(prop,seconds)local s=propState[prop];if not s or not s.Nailed or s.Infected then return end;s.Infected=true;sync(prop);task.spawn(function(local left=seconds while left>0 and propState[prop]and propState[prop].Nailed do task.wait(1);left-=1;BarricadeManager.DamageProp(prop,Config.SPIT_DOT,nil)end;if propState[prop]then propState[prop].Infected=false;sync(prop)end end)end
function BarricadeManager.IsNailedProp(i)return propState[i]~=nil and propState[i].Nailed end
function BarricadeManager.ResolveProp(i)local c=i;while c and c~=Workspace do if propState[c]then return c end;c=c.Parent end end
function BarricadeManager.GetNailedInSector(_)local out={};for p,s in propState do if s.Nailed and p.Parent then table.insert(out,p)end end;return out end
function BarricadeManager.ResetMap()for p in carryByPlayer do release(p)end;for p in propState do if p.Parent then p:Destroy()end end;table.clear(propState);table.clear(carrierByProp);table.clear(placedByPlayer);table.clear(dismantling);placedTotal=0;PropClutter.ClearAll();local templates=ServerStorage:FindFirstChild("PropTemplates");local spawns=Workspace:FindFirstChild("PropSpawns");if not templates or not spawns then return end;task.spawn(function()local made=0;for _,marker in spawns:GetDescendants()do if marker:IsA("BasePart")then local kind=tostring(marker:GetAttribute("PropType")or"WoodCrate");local tpl=templates:FindFirstChild(kind);if tpl then local c=tpl:Clone();c:SetAttribute("PropType",kind);c.Parent=propsFolder;Placement.ApplyCenter(c,marker.CFrame);BarricadeManager.RegisterProp(c);made+=1;if made%12==0 then task.wait()end end end end end)end
local function carryStep()for p,c in carryByPlayer do local r=AntiCheat.GetRoot(p);if not r or not c.Prop.Parent then release(p)elseif Placement.ReachDistance(r.Position,c.Prop)>B.MaxCarryDistance then onDrop(p)else local look=r.CFrame.LookVector;local yaw=math.atan2(-look.X,-look.Z);Placement.ApplyCenter(c.Prop,CFrame.new(r.Position)*CFrame.Angles(0,yaw,0)*CFrame.new(B.CarryOffset))end end end
local function dismantleStep()local now=os.clock();for p,j in dismantling do local prop=j.Prop;if not p.Parent or not prop.Parent or not ready(p)or not inReach(p,prop,H.Reach)or not BarricadeManager.IsNailedProp(prop)then BarricadeManager.CancelDismantle(p)elseif j.Until<=now then dismantling[p]=nil;Remotes.ReviveProgress:FireClient(p,0);finish(p,prop)else Remotes.ReviveProgress:FireClient(p,1-(j.Until-now)/H.DismantleTime)end end end
function BarricadeManager.Init()propsFolder=Workspace:FindFirstChild("BarricadeProps")or Instance.new("Folder");propsFolder.Name="BarricadeProps";propsFolder.Parent=Workspace;local sockets=Workspace:FindFirstChild("BarricadeSockets");if sockets then for _,s in sockets:GetDescendants()do BarricadeManager.RegisterSocket(s)end end;for _,i in propsFolder:GetChildren()do BarricadeManager.RegisterProp(i)end;AntiCheat.Bind(Remotes.PickupRequest,"PickupRequest",onPickup);AntiCheat.Bind(Remotes.DropRequest,"DropRequest",onDrop);AntiCheat.Bind(Remotes.PlaceFreeRequest,"PlaceFreeRequest",onPlace);AntiCheat.Bind(Remotes.RepairRequest,"RepairRequest",function(p,prop)BarricadeManager.Repair(p,prop)end);AntiCheat.Bind(Remotes.PlaceRequest,"PlaceRequest",function()end);RunService.Heartbeat:Connect(function()local ok,e=pcall(carryStep);if not ok then warn("[Barricade] carryStep: "..tostring(e))end;local ok2,e2=pcall(dismantleStep);if not ok2 then warn("[Barricade] dismantle: "..tostring(e2))end end);Players.PlayerRemoving:Connect(function(p)release(p);repairCd[p]=nil;dismantling[p]=nil end);local function hook(p)p.CharacterRemoving:Connect(function()release(p);dismantling[p]=nil end)end;Players.PlayerAdded:Connect(hook);for _,p in Players:GetPlayers()do hook(p)end;task.defer(function()if #propsFolder:GetChildren()==0 then pcall(BarricadeManager.ResetMap)end end)end
return BarricadeManager
