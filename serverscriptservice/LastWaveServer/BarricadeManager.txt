local Players=game:GetService("Players")
local Workspace=game:GetService("Workspace")
local ServerStorage=game:GetService("ServerStorage")
local RunService=game:GetService("RunService")
local CollectionService=game:GetService("CollectionService")
local Shared=game.ReplicatedStorage.LastWaveShared
local Config=require(Shared.Config);local Remotes=require(Shared.Remotes);local Sanity=require(Shared.Sanity);local PropTypes=require(Shared.PropTypes);local Kits=require(Shared.SurvivorKits);local Placement=require(Shared.PlacementRules)
local AntiCheat=require(script.Parent.AntiCheat);local PhysicsSetup=require(script.Parent.PhysicsSetup);local Economy=require(script.Parent.EconomyManager);local Score=require(script.Parent.ScoreManager);local Match=require(script.Parent.MatchState);local Clutter=require(script.Parent.PropClutter)
local M={};local B=Config.BARRICADE;local H=Config.HAMMER;local state={};local carry={};local carrier={};local repairCd={};local dismantle={};local placedBy={};local total=0;local folder
local function part(p)if p:IsA("BasePart")then return p end;return p:IsA("Model")and(p.PrimaryPart or p:FindFirstChildWhichIsA("BasePart"))end
local function each(p,fn)if p:IsA("BasePart")then fn(p)end;for _,x in p:GetDescendants()do if x:IsA("BasePart")then fn(x)end end end
local function anchor(p,v)each(p,function(x)x.Anchored=v end)end
local function collide(p,v)each(p,function(x)x.CanCollide=v end)end
local function ready(pl)local c=pl.Character;local h=c and c:FindFirstChildWhichIsA("Humanoid");return c and c:GetAttribute("Role")=="Survivor"and not c:GetAttribute("Downed")and h and h.Health>0 end
local function reach(pl,p,d)local r=AntiCheat.GetRoot(pl);return r and Placement.ReachDistance(r.Position,p)<=d end
local function mark(pl,p)local c=pl.Character;if not c then return end;local v=c:FindFirstChild("CarriedProp")or Instance.new("ObjectValue");v.Name="CarriedProp";v.Parent=c;v.Value=p end
local function sync(p)local s=state[p];if not s then return end;p:SetAttribute("HP",math.floor(s.HP));p:SetAttribute("MaxHP",math.floor(s.MaxHP));p:SetAttribute("IsNailed",s.Nailed);p:SetAttribute("Infected",s.Infected);p:SetAttribute("PlacedBy",s.PlacedBy)end
local function kit(pl)local c=pl.Character;return Kits[c and c:GetAttribute("Kit")or"Builder"]or Kits.Builder end
local function settle(p)local s=Placement.SizeOf(p);local cf=p:IsA("Model")and p:GetBoundingBox()or p.CFrame;local rp=RaycastParams.new();rp.FilterType=Enum.RaycastFilterType.Exclude;rp.FilterDescendantsInstances={p};local h=Workspace:Raycast(cf.Position+Vector3.new(0,2,0),Vector3.new(0,-200,0),rp);if h then Placement.ApplyCenter(p,CFrame.new(h.Position+Vector3.new(0,s.Y*.5+.02,0))*CFrame.Angles(0,math.atan2(-cf.LookVector.X,-cf.LookVector.Z),0))end end
function M.RegisterProp(p)local q=part(p);if not q then return end;local k=tostring(p:GetAttribute("PropType")or p.Name);local d=PropTypes[k]or PropTypes.WoodCrate;if not PropTypes[k]then k="WoodCrate"end;local s=p:IsA("Model")and select(2,p:GetBoundingBox())or q.Size;p:SetAttribute("PropType",k);p:SetAttribute("PropSize",s);state[p]={HP=d.MaxHP,MaxHP=d.MaxHP,BaseMaxHP=d.MaxHP,Nailed=false,Infected=false,PlacedBy=0,PropType=k,Size=s};CollectionService:AddTag(p,"BarricadeProp");PhysicsSetup.SetPropGroup(p,false);Clutter.Register(p);anchor(p,true);collide(p,true);settle(p);sync(p)end
function M.RegisterSocket(s)if s:IsA("BasePart")then s.Anchored=true;s.CanCollide=false;s.Transparency=1 end end
function M.GetFreeSockets()return{}end
local function release(pl,keep)local c=carry[pl];if not c then return end;carry[pl]=nil;carrier[c.Prop]=nil;if c.Prop.Parent then c.Prop:SetAttribute("CarriedBy",nil);if not keep then PhysicsSetup.SetPropGroup(c.Prop,false);collide(c.Prop,true)end end;if pl.Character then pl.Character:SetAttribute("CarryMultiplier",1)end;mark(pl,nil);return c end
function M.TakeIntoHands(pl,p)if carry[pl]then return false end;local s=state[p];if not s or s.Nailed or carrier[p]then return false end;local d=PropTypes[s.PropType];carry[pl]={Prop=p,Def=d};carrier[p]=pl;anchor(p,true);collide(p,false);PhysicsSetup.SetPropGroup(p,true);p:SetAttribute("CarriedBy",pl.UserId);mark(pl,p);Clutter.Release(p);local c=pl.Character;c:SetAttribute("CarryMultiplier",math.clamp(d.CarrySpeedMultiplier+kit(pl).CarryBonus,.35,1));return true end
local function pickup(pl,p)
 if not Sanity.inst(p,"Instance")or not ready(pl)or carry[pl]then return end
 local Hammer=require(script.Parent.HammerManager);if not Hammer.IsEquipped(pl)then Remotes.Notify:FireClient(pl,"needHammer");return end
 local s=state[p];if not s or not CollectionService:HasTag(p,"BarricadeProp")or not p:IsDescendantOf(folder)then return end;if s.Nailed then Remotes.Notify:FireClient(pl,"nailedHint");return end;if reach(pl,p,Config.AC.MaxInteractDistance)then M.TakeIntoHands(pl,p)end
end
local function drop(pl)local c=release(pl);if not c or not c.Prop.Parent then return end;local r=AntiCheat.GetRoot(pl);if r then Placement.ApplyCenter(c.Prop,CFrame.new(r.Position+r.CFrame.LookVector*4));settle(c.Prop)end end
local function place(pl,dir,yaw)
 if not Sanity.unitVec(dir)or not ready(pl)or Match.Phase()=="GameOver"then return end;local c=carry[pl];local h=pl.Character:FindFirstChild("Head")or pl.Character:FindFirstChild("HumanoidRootPart");if not c or not h then return end;local limit=B.LimitPerPlayer+kit(pl).ExtraPropLimit;if(placedBy[pl.UserId]or 0)>=limit or total>=B.LimitTotal then Remotes.Notify:FireClient(pl,"propLimit");return end;local cf,why=Placement.Solve({origin=h.Position,dir=dir,maxDist=B.MaxPlaceDistance,size=state[c.Prop].Size,yaw=type(yaw)=="number"and yaw or 0,ignore={pl.Character,c.Prop}});if not cf then Remotes.Notify:FireClient(pl,"placeFail",why);return end;if next(c.Def.PlaceCost)and not Economy.Spend(pl,c.Def.PlaceCost)then Remotes.Notify:FireClient(pl,"noResources");return end;release(pl,true);Placement.ApplyCenter(c.Prop,cf);anchor(c.Prop,true);collide(c.Prop,true);PhysicsSetup.SetPropGroup(c.Prop,false);local s=state[c.Prop];s.Nailed=true;s.PlacedBy=pl.UserId;placedBy[pl.UserId]=(placedBy[pl.UserId]or 0)+1;total+=1;sync(c.Prop);Score.Award(pl,"PropPlaced")
end
function M.Repair(pl,p)if not ready(pl)or not Sanity.inst(p,"Instance")then return false end;local s=state[p];if not s or not s.Nailed or s.Infected or s.HP>=s.MaxHP or not reach(pl,p,H.Reach)then return false end;local now=os.clock();if(repairCd[pl]or 0)+H.RepairCooldown>now or not Economy.Spend(pl,H.RepairCost)then return false end;repairCd[pl]=now;local k=kit(pl);s.MaxHP=math.max(s.BaseMaxHP*H.MinMaxHPRatio,s.MaxHP-s.BaseMaxHP*H.DegradePerRepair);s.HP=math.clamp(s.HP+H.RepairAmount*k.RepairMultiplier,0,s.MaxHP);sync(p);Score.Award(pl,"PropRepaired");return true end
function M.BeginDismantle(pl,p)if not ready(pl)or not Sanity.inst(p,"Instance")then return end;local s=state[p];if not s or not s.Nailed or not reach(pl,p,H.Reach)then return end;if H.OwnOnly and s.PlacedBy~=pl.UserId then return end;dismantle[pl]={Prop=p,Until=os.clock()+H.DismantleTime}end
function M.CancelDismantle(pl)if dismantle[pl]then dismantle[pl]=nil;Remotes.ReviveProgress:FireClient(pl,0)end end
local function unNail(p)local s=state[p];if not s or not s.Nailed then return end;s.Nailed=false;if s.PlacedBy~=0 then placedBy[s.PlacedBy]=math.max(0,(placedBy[s.PlacedBy]or 1)-1)end;s.PlacedBy=0;total=math.max(0,total-1);sync(p)end
local function finish(pl,p)local s=state[p];if not s or not s.Nailed then return end;local d=PropTypes[s.PropType];unNail(p);if next(d.PlaceCost)and H.DismantleRefund>0 then local back={};for k,v in d.PlaceCost do back[k]=math.max(1,math.floor(v*H.DismantleRefund))end;Economy.Add(pl,back)end;if H.DismantleToHands and not carry[pl]then M.TakeIntoHands(pl,p)else settle(p)end;Remotes.Notify:FireClient(pl,"dismantled")end
function M.DamageProp(p,n,src)local s=state[p];if not s or not s.Nailed or not Sanity.num(n,0,100000)then return end;s.HP=math.clamp(s.HP-n,0,s.MaxHP);sync(p);if s.HP>0 then return end;if src then Score.Award(src,"PropDestroyed")end;unNail(p);collide(p,false);task.delay(Config.PROP_DESPAWN_TIME,function()Clutter.Forget(p);if p.Parent then p:Destroy()end;state[p]=nil;carrier[p]=nil end)end
function M.InfectProp(p,seconds)local s=state[p];if not s or not s.Nailed or s.Infected then return end;s.Infected=true;sync(p);task.spawn(function()local left=seconds;while left>0 and state[p]and state[p].Nailed do task.wait(1);left-=1;M.DamageProp(p,Config.SPIT_DOT,nil)end;if state[p]then state[p].Infected=false;sync(p)end end)end
function M.IsNailedProp(p)return state[p]~=nil and state[p].Nailed end
function M.ResolveProp(i)local c=i;while c and c~=Workspace do if state[c]then return c end;c=c.Parent end end
function M.GetNailedInSector(_)local out={};for p,s in state do if s.Nailed and p.Parent then table.insert(out,p)end end;return out end
function M.ResetMap()for p in carry do release(p)end;for p in state do if p.Parent then p:Destroy()end end;table.clear(state);table.clear(carrier);table.clear(placedBy);table.clear(dismantle);total=0;Clutter.ClearAll();local t=ServerStorage:FindFirstChild("PropTemplates");local sp=Workspace:FindFirstChild("PropSpawns");if not t or not sp then return end;for _,m in sp:GetDescendants()do if m:IsA("BasePart")then local k=tostring(m:GetAttribute("PropType")or"WoodCrate");local x=t:FindFirstChild(k);if x then local p=x:Clone();p:SetAttribute("PropType",k);p.Parent=folder;Placement.ApplyCenter(p,m.CFrame);M.RegisterProp(p)end end end end
function M.Init()folder=Workspace:FindFirstChild("BarricadeProps")or Instance.new("Folder");folder.Name="BarricadeProps";folder.Parent=Workspace;for _,p in folder:GetChildren()do M.RegisterProp(p)end;AntiCheat.Bind(Remotes.PickupRequest,"PickupRequest",pickup);AntiCheat.Bind(Remotes.DropRequest,"DropRequest",drop);AntiCheat.Bind(Remotes.PlaceFreeRequest,"PlaceFreeRequest",place);AntiCheat.Bind(Remotes.RepairRequest,"RepairRequest",function(p,x)M.Repair(p,x)end);RunService.Heartbeat:Connect(function()for p,c in carry do local r=AntiCheat.GetRoot(p);if not r or not c.Prop.Parent then release(p)else Placement.ApplyCenter(c.Prop,CFrame.new(r.Position)*CFrame.Angles(0,math.atan2(-r.CFrame.LookVector.X,-r.CFrame.LookVector.Z),0)*CFrame.new(B.CarryOffset))end end;local now=os.clock();for p,j in dismantle do if not ready(p)or not j.Prop.Parent or not reach(p,j.Prop,H.Reach)then M.CancelDismantle(p)elseif j.Until<=now then dismantle[p]=nil;finish(p,j.Prop)else Remotes.ReviveProgress:FireClient(p,1-(j.Until-now)/H.DismantleTime)end end end)end
return M
