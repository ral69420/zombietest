function HammerManager.Give(p:Player)
 if not H.Enabled then return end;local c=p.Character;local b=p:FindFirstChildOfClass("Backpack");if not c or not b or c:FindFirstChild(NAME)or b:FindFirstChild(NAME)then return end;local t=template();t.Parent=b;c:SetAttribute("HammerEquipped",false)

end
