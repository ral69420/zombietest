--!nonstrict
local Players=game:GetService("Players")
local StarterGui=game:GetService("StarterGui")
local ServerStorage=game:GetService("ServerStorage")
local Shared=game.ReplicatedStorage:WaitForChild("LastWaveShared")
local Config=require(Shared.Config);local Remotes=require(Shared.Remotes);local Sanity=require(Shared.Sanity)
local AntiCheat=require(script.Parent.AntiCheat);local BarricadeManager=require(script.Parent.BarricadeManager);local WeaponRig=require(script.Parent.WeaponRig)
local HammerManager={};local H=Config.HAMMER;local NAME="Hammer"

local function dropCarried(p:Player)if H.DropPropOnUnequip~=false then BarricadeManager.ForceDrop(p)end end

local function defaultModel():Model
 local m=Instance.new("Model");m.Name=NAME
 local h=Instance.new("Part");h.Name="Handle";h.Size=Vector3.new(.3,2,.3);h.Color=Color3.fromRGB(120,85,50);h.CanCollide=false;h.CanTouch=false;h.CanQuery=false;h.Massless=true;h.Parent=m
 local head=Instance.new("Part");head.Name="HammerHead";head.Size=Vector3.new(.55,.5,1.1);head.Color=Color3.fromRGB(150,150,162);head.CanCollide=false;head.CanTouch=false;head.CanQuery=false;head.Massless=true;head.CFrame=h.CFrame*CFrame.new(0,.95,0);head.Parent=m
 local w=Instance.new("WeldConstraint");w.Part0=h;w.Part1=head;w.Parent=h
 m.PrimaryPart=h
 return m
end

local function template():Model
 local f=ServerStorage:FindFirstChild("ToolModels")
 local t=f and f:FindFirstChild(H.ModelName)
 if t then
  local c=t:Clone()
  c.Name=NAME
  if c:IsA("Tool") then
   local m=Instance.new("Model")
   m.Name=NAME
   for _, child in c:GetChildren() do child.Parent=m end
   m.PrimaryPart=m:FindFirstChild("Handle") or m:FindFirstChildWhichIsA("BasePart")
   return m
  elseif c:IsA("Model") then
   c.PrimaryPart=c.PrimaryPart or c:FindFirstChild("Handle") or c:FindFirstChildWhichIsA("BasePart")
   return c
  end
 end
 return defaultModel()
end

local function detachModel(c:Model)
 local old=c:FindFirstChild("HammerModel")
 if old then old:Destroy() end
 local g=c:FindFirstChild("HammerGrip",true)
 if g then g:Destroy() end
end

local function attachModel(p:Player)
 local c=p.Character;if not c then return end
 detachModel(c)
 local arm=c:FindFirstChild("Right Arm") or c:FindFirstChild("RightHand")
 if not arm or not arm:IsA("BasePart") then return end
 local m=template()
 m.Name="HammerModel"
 for _, pt in m:GetDescendants() do
  if pt:IsA("BasePart") then
   pt.Anchored=false;pt.CanCollide=false;pt.CanQuery=false;pt.CanTouch=false;pt.Massless=true;pt.CastShadow=false
   pt.CollisionGroup=Config.GROUP_SURVIVOR
  end
 end
 local root=m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart")
 if not root then return end
 m.Parent=c
 root.CFrame=arm.CFrame*H.Grip
 local w=Instance.new("WeldConstraint")
 w.Name="HammerGrip"
 w.Part0=arm
 w.Part1=root
 w.Parent=arm
end

local function restoreWeapon(p:Player)
 local wm=require(script.Parent.WeaponManager)
 local id=wm.GetId(p)
 if id then WeaponRig.Attach(p,id) end
end

function HammerManager.Give(p:Player)
 if not H.Enabled then return end
 local c=p.Character;if not c then return end
 c:SetAttribute("HasHammer",true)
 c:SetAttribute("HammerEquipped",false)
end

function HammerManager.Remove(p:Player)
 local c=p.Character
 if c then
  c:SetAttribute("HasHammer",false)
  c:SetAttribute("HammerEquipped",false)
  detachModel(c)
 end
 BarricadeManager.CancelDismantle(p)
 dropCarried(p)
end

function HammerManager.Equip(p:Player, equip:boolean)
 local c=p.Character;if not c or c:GetAttribute("Role")~="Survivor" or c:GetAttribute("Downed") then return end
 if equip then
  if not c:GetAttribute("HasHammer") then return end
  c:SetAttribute("HammerEquipped",true)
  WeaponRig.Detach(c)
  attachModel(p)
  BarricadeManager.CancelDismantle(p)
 else
  if c:GetAttribute("HammerEquipped")==true then
   c:SetAttribute("HammerEquipped",false)
   detachModel(c)
   BarricadeManager.CancelDismantle(p)
   dropCarried(p)
   restoreWeapon(p)
  end
 end
end

function HammerManager.IsEquipped(p:Player):boolean
 local c=p.Character
 return c~=nil and c:GetAttribute("HammerEquipped")==true
end

function HammerManager.HasHammer(p:Player):boolean
 local c=p.Character
 return c~=nil and c:GetAttribute("HasHammer")==true
end

function HammerManager.Init()
 AntiCheat.Bind(Remotes.HammerRepairRequest,"HammerRepairRequest",function(p,prop)
  if HammerManager.IsEquipped(p) and Sanity.inst(prop,"Instance") then BarricadeManager.Repair(p,prop) end
 end)
 AntiCheat.Bind(Remotes.HammerDismantleRequest,"HammerDismantleRequest",function(p,prop,start)
  if type(start)~="boolean" then return end
  if not start then BarricadeManager.CancelDismantle(p);return end
  if Sanity.inst(prop,"Instance") then BarricadeManager.BeginDismantle(p,prop) end
 end)
 Players.PlayerRemoving:Connect(function(p)BarricadeManager.CancelDismantle(p)end)
end
return HammerManager
