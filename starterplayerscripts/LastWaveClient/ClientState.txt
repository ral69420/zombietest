--!nonstrict
local Players=game:GetService("Players")
local UserInputService=game:GetService("UserInputService")
local ContextActionService=game:GetService("ContextActionService")
local Workspace=game:GetService("Workspace")
local Config=require(game.ReplicatedStorage.LastWaveShared.Config)
local ClientState={};local player=Players.LocalPlayer;local locked=false;local saved=nil;local ACTION="LastWaveShopBlock"
ClientState.Changed=Instance.new("BindableEvent")
function ClientState.IsUILocked()return locked end
function ClientState.ApplyMouseIcon()if Config.MOUSE_ICON and Config.MOUSE_ICON~=""then player:GetMouse().Icon=Config.MOUSE_ICON end end
local function sink()return Enum.ContextActionResult.Sink end
function ClientState.SetUILock(on:boolean)
 if on==locked then return end;locked=on;local camera=Workspace.CurrentCamera;local hum=player.Character and player.Character:FindFirstChildWhichIsA("Humanoid")
 if on then saved={behavior=UserInputService.MouseBehavior,iconEnabled=UserInputService.MouseIconEnabled,autoRotate=hum and hum.AutoRotate,walkSpeed=hum and hum.WalkSpeed,jumpPower=hum and hum.JumpPower};ContextActionService:BindActionAtPriority(ACTION,sink,false,Enum.ContextActionPriority.High.Value,Enum.PlayerActions.CharacterForward,Enum.PlayerActions.CharacterBackward,Enum.PlayerActions.CharacterLeft,Enum.PlayerActions.CharacterRight,Enum.PlayerActions.CharacterJump);UserInputService.MouseBehavior=Enum.MouseBehavior.Default;UserInputService.MouseIconEnabled=true;ClientState.ApplyMouseIcon();if camera then camera.CameraType=Enum.CameraType.Scriptable end;if hum then hum.AutoRotate=false;hum.WalkSpeed=0;hum.JumpPower=0 end
 else ContextActionService:UnbindAction(ACTION);UserInputService.MouseBehavior=(saved and saved.behavior)or Enum.MouseBehavior.LockCenter;UserInputService.MouseIconEnabled=(saved and saved.iconEnabled)or false;if camera then camera.CameraType=Enum.CameraType.Scriptable end;if hum then hum.AutoRotate=(saved and saved.autoRotate)~=false;hum.WalkSpeed=saved and saved.walkSpeed or hum.WalkSpeed;hum.JumpPower=saved and saved.jumpPower or hum.JumpPower end;saved=nil end
 ClientState.Changed:Fire(locked)
end
function ClientState.Init()ClientState.ApplyMouseIcon();player.CharacterAdded:Connect(function()task.wait(.2);ClientState.ApplyMouseIcon();if locked then local hum=player.Character and player.Character:FindFirstChildWhichIsA("Humanoid");if hum then hum.WalkSpeed=0;hum.JumpPower=0 end;UserInputService.MouseBehavior=Enum.MouseBehavior.Default;UserInputService.MouseIconEnabled=true end end)end
return ClientState
