--!nonstrict
-- Мобильное управление. Стильные анимированные тач-кнопки в стиле Build Simple / PS1
local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local TweenService = game:GetService("TweenService")
local CollectionService = game:GetService("CollectionService")

local Remotes = require(game.ReplicatedStorage.LastWaveShared.Remotes)
local BarricadeController = require(script.Parent.BarricadeController)
local WeaponController = require(script.Parent.WeaponController)
local MovementController = require(script.Parent.MovementController)
local ShopController = require(script.Parent.ShopController)

local MobileController = {}
local localPlayer = Players.LocalPlayer

local showInfo = TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local pressInfo = TweenInfo.new(0.06, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local backInfo = TweenInfo.new(0.10, Enum.EasingStyle.Back, Enum.EasingDirection.Out)

local C_TAKE = Color3.fromRGB(58, 130, 58)
local C_CANCEL = Color3.fromRGB(160, 50, 50)
local C_PLACE = Color3.fromRGB(50, 100, 200)
local C_ROT = Color3.fromRGB(90, 90, 90)
local C_FIX = Color3.fromRGB(160, 140, 60)
local C_REMOVE = Color3.fromRGB(150, 80, 60)

local function makeBtn(gui, label, subLabel, pos, bgColor, borderCol, size)
	local frame = Instance.new("Frame")
	frame.Size = UDim2.fromOffset(size, size)
	frame.AnchorPoint = Vector2.new(0.5, 0.5)
	frame.Position = pos
	frame.BackgroundColor3 = bgColor
	frame.BorderSizePixel = 0
	frame.ZIndex = 10
	frame.BackgroundTransparency = 0.15
	frame.Parent = gui

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(1, 0)
	corner.Parent = frame

	local stroke = Instance.new("UIStroke")
	stroke.Thickness = 1.5
	stroke.Color = borderCol
	stroke.Transparency = 0.2
	stroke.Parent = frame

	local mainLbl = Instance.new("TextLabel")
	mainLbl.Size = UDim2.fromScale(1, 0.55)
	mainLbl.Position = UDim2.fromScale(0, 0.08)
	mainLbl.BackgroundTransparency = 1
	mainLbl.Text = label
	mainLbl.Font = Enum.Font.GothamBold
	mainLbl.TextScaled = true
	mainLbl.TextColor3 = Color3.new(1, 1, 1)
	mainLbl.ZIndex = 11
	mainLbl.Parent = frame

	local subLbl = Instance.new("TextLabel")
	subLbl.Size = UDim2.new(1, -4, 0, 14)
	subLbl.AnchorPoint = Vector2.new(0.5, 1)
	subLbl.Position = UDim2.new(0.5, 0, 1, -4)
	subLbl.BackgroundTransparency = 1
	subLbl.Text = subLabel
	subLbl.Font = Enum.Font.Gotham
	subLbl.TextSize = 10
	subLbl.TextColor3 = Color3.fromRGB(210, 210, 210)
	subLbl.ZIndex = 11
	subLbl.Parent = frame

	local tb = Instance.new("TextButton")
	tb.Size = UDim2.fromScale(1, 1)
	tb.BackgroundTransparency = 1
	tb.Text = ""
	tb.ZIndex = 12
	tb.Parent = frame

	return frame, tb, mainLbl, subLbl
end

local function showFrame(frame: Frame, bgColor: Color3, borderCol: Color3)
	local stroke = frame:FindFirstChildOfClass("UIStroke")
	frame.Visible = true
	TweenService:Create(frame, showInfo, { BackgroundTransparency = 0.15, BackgroundColor3 = bgColor }):Play()
	if stroke then TweenService:Create(stroke, showInfo, { Transparency = 0.2, Color = borderCol }):Play() end
	for _, ch in ipairs(frame:GetChildren()) do
		if ch:IsA("TextLabel") then
			TweenService:Create(ch, showInfo, { TextTransparency = 0 }):Play()
		end
	end
end

local function hideFrame(frame: Frame)
	local stroke = frame:FindFirstChildOfClass("UIStroke")
	local tw = TweenService:Create(frame, showInfo, { BackgroundTransparency = 1 })
	tw:Play()
	if stroke then TweenService:Create(stroke, showInfo, { Transparency = 1 }):Play() end
	for _, ch in ipairs(frame:GetChildren()) do
		if ch:IsA("TextLabel") then
			TweenService:Create(ch, showInfo, { TextTransparency = 1 }):Play()
		end
	end
	task.delay(0.15, function()
		if frame.BackgroundTransparency >= 0.95 then frame.Visible = false end
	end)
end

local function bindPress(frame: Frame, tb: TextButton, normalColor: Color3, normalSize: number, actionDown: () -> (), actionUp: (() -> ())?)
	local function onDown()
		TweenService:Create(frame, pressInfo, {
			BackgroundColor3 = normalColor:Lerp(Color3.new(0, 0, 0), 0.35),
			Size = UDim2.fromOffset(normalSize - 8, normalSize - 8),
		}):Play()
		actionDown()
	end
	local function onUp()
		TweenService:Create(frame, backInfo, {
			BackgroundColor3 = normalColor,
			Size = UDim2.fromOffset(normalSize, normalSize),
		}):Play()
		if actionUp then actionUp() end
	end

	tb.MouseButton1Down:Connect(onDown)
	tb.MouseButton1Up:Connect(onUp)
	tb.MouseLeave:Connect(onUp)
	tb.TouchTap:Connect(function()
		onDown()
		task.delay(0.08, onUp)
	end)
end

function MobileController.Init()
	if not UserInputService.TouchEnabled then return end

	local gui = Instance.new("ScreenGui")
	gui.Name = "LastWaveTouch"
	gui.ResetOnSpawn = false
	gui.IgnoreGuiInset = true
	gui.DisplayOrder = 15
	gui.Parent = localPlayer:WaitForChild("PlayerGui")

	local BTN_MAIN = 74
	local BTN_SEC = 56

	local frameE, tbE, lblE, subE = makeBtn(gui, "E", "take", UDim2.new(1, -95, 1, -140), C_TAKE, C_TAKE:Lerp(Color3.new(1, 1, 1), 0.3), BTN_MAIN)
	local frameY, tbY = makeBtn(gui, "Y", "place", UDim2.new(1, -20, 1, -140), C_PLACE, C_PLACE:Lerp(Color3.new(1, 1, 1), 0.3), BTN_MAIN)
	local frameR, tbR = makeBtn(gui, "R", "rotate", UDim2.new(1, -95, 1, -225), C_ROT, C_ROT:Lerp(Color3.new(1, 1, 1), 0.3), BTN_SEC)
	local frameC, tbC = makeBtn(gui, "✕", "cancel", UDim2.new(1, -20, 1, -225), C_CANCEL, C_CANCEL:Lerp(Color3.new(1, 1, 1), 0.3), BTN_SEC)

	local frameFire, tbFire = makeBtn(gui, "FIRE", "shoot", UDim2.new(1, -95, 1, -315), Color3.fromRGB(180, 50, 50), Color3.fromRGB(240, 90, 90), BTN_MAIN)
	local frameFix, tbFix = makeBtn(gui, "FIX", "repair", UDim2.new(1, -175, 1, -140), C_FIX, C_FIX:Lerp(Color3.new(1,1,1), 0.3), BTN_SEC)
	local frameRemove, tbRemove = makeBtn(gui, "REM", "dismantle", UDim2.new(1, -175, 1, -225), C_REMOVE, C_REMOVE:Lerp(Color3.new(1,1,1), 0.3), BTN_SEC)
	local frameSprint, tbSprint = makeBtn(gui, "SPRINT", "run", UDim2.new(0, 80, 1, -140), Color3.fromRGB(70, 130, 200), Color3.fromRGB(120, 180, 255), BTN_MAIN)

	bindPress(frameE, tbE, C_TAKE, BTN_MAIN, function() BarricadeController.TryPickupOrDrop() end)
	bindPress(frameY, tbY, C_PLACE, BTN_MAIN, function() BarricadeController.TryPlace() end)
	bindPress(frameR, tbR, C_ROT, BTN_SEC, function() BarricadeController.Rotate(1, false) end)
	bindPress(frameC, tbC, C_CANCEL, BTN_SEC, function() BarricadeController.TryPickupOrDrop() end)
	bindPress(frameFire, tbFire, Color3.fromRGB(180, 50, 50), BTN_MAIN, function()
		local char = localPlayer.Character
		if char and char:GetAttribute("Role") == "Zombie" then
			BarricadeController.Attack()
		else
			WeaponController.Fire()
		end
	end)
	bindPress(frameFix, tbFix, C_FIX, BTN_SEC, function() BarricadeController.HammerHit() end)
	bindPress(frameRemove, tbRemove, C_REMOVE, BTN_SEC,
		function() BarricadeController.HammerDismantle(true) end,
		function() BarricadeController.HammerDismantle(false) end
	)
	bindPress(frameSprint, tbSprint, Color3.fromRGB(70, 130, 200), BTN_MAIN,
		function() MovementController.SetSprint(true) end,
		function() MovementController.SetSprint(false) end
	)

	RunService.Heartbeat:Connect(function()
		local char = localPlayer.Character
		local role = char and char:GetAttribute("Role")
		local isSurv = role == "Survivor"
		local carrying = BarricadeController.IsCarrying()
		local isHammer = char and char:GetAttribute("HammerEquipped") == true

		if isSurv then
			if carrying then
				lblE.Text = "DROP"
				subE.Text = "drop"
				showFrame(frameY, C_PLACE, C_PLACE:Lerp(Color3.new(1,1,1), 0.3))
				showFrame(frameR, C_ROT, C_ROT:Lerp(Color3.new(1,1,1), 0.3))
				showFrame(frameC, C_CANCEL, C_CANCEL:Lerp(Color3.new(1,1,1), 0.3))
				hideFrame(frameFix)
				hideFrame(frameRemove)
			else
				lblE.Text = "E"
				subE.Text = "take"
				hideFrame(frameY)
				hideFrame(frameR)
				hideFrame(frameC)
				if isHammer then
					showFrame(frameFix, C_FIX, C_FIX:Lerp(Color3.new(1,1,1), 0.3))
					showFrame(frameRemove, C_REMOVE, C_REMOVE:Lerp(Color3.new(1,1,1), 0.3))
				else
					hideFrame(frameFix)
					hideFrame(frameRemove)
				end
			end
		else
			hideFrame(frameY)
			hideFrame(frameR)
			hideFrame(frameC)
			hideFrame(frameFix)
			hideFrame(frameRemove)
			lblE.Text = "E"
			subE.Text = "interact"
		end
	end)
end

return MobileController
