--!strict
local Players=game:GetService("Players")
local RunService=game:GetService("RunService")
local UserInputService=game:GetService("UserInputService")
local Workspace=game:GetService("Workspace")
local CameraController={};local player=Players.LocalPlayer;local camera=Workspace.CurrentCamera;local ClientState=require(script.Parent.ClientState)
local yaw,pitch=0,0;local renderConn:RBXScriptConnection?;local stateConn:RBXScriptConnection?;local BASE_FOV=90
local function resetCamera()if renderConn then renderConn:Disconnect();renderConn=nil end;if stateConn then stateConn:Disconnect();stateConn=nil end;camera=Workspace.CurrentCamera;if camera then camera.CameraType=Enum.CameraType.Custom;camera.CameraSubject=nil end end
local function setup(char:Model)
 resetCamera();local hum=char:WaitForChild("Humanoid",10)::Humanoid?;local root=char:WaitForChild("HumanoidRootPart",10)::BasePart?;local head=char:FindFirstChild("Head")::BasePart?;if not hum or not root then return end;head=head or root;camera=Workspace.CurrentCamera;hum.AutoRotate=true;if not ClientState.IsUILocked()then UserInputService.MouseBehavior=Enum.MouseBehavior.LockCenter;UserInputService.MouseIconEnabled=false end;if camera then camera.CameraType=Enum.CameraType.Scriptable;camera.FieldOfView=BASE_FOV end
 stateConn=hum.Died:Connect(resetCamera)
 renderConn=RunService.RenderStepped:Connect(function(dt)
  if not char.Parent or not root.Parent or hum.Health<=0 then resetCamera();return end;if ClientState.IsUILocked()then return end;UserInputService.MouseBehavior=Enum.MouseBehavior.LockCenter;UserInputService.MouseIconEnabled=false;local delta=UserInputService:GetMouseDelta();yaw=(yaw-delta.X*.18)%360;pitch=math.clamp(pitch-delta.Y*.18,-80,80);local moving=root.AssemblyLinearVelocity*Vector3.new(1,0,1);local fov=BASE_FOV+(moving.Magnitude>1 and 8 or 0);camera.CFrame=CFrame.new(head.Position)*CFrame.Angles(0,math.rad(yaw),0)*CFrame.Angles(math.rad(pitch),0,0)*CFrame.new(0,.35,0);camera.FieldOfView=camera.FieldOfView+(fov-camera.FieldOfView)*math.min(dt*8,1);root.CFrame=CFrame.new(root.Position)*CFrame.Angles(0,math.rad(yaw),0)
 end)
end
function CameraController.Init()
 player.CharacterAdded:Connect(setup);player.CharacterRemoving:Connect(resetCamera);ClientState.Changed.Event:Connect(function(locked)if locked then if camera then camera.CameraType=Enum.CameraType.Scriptable end else task.defer(function()if player.Character then setup(player.Character)end end)end end);if player.Character then setup(player.Character)end
end
return CameraController
