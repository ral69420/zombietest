--!strict
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local Workspace = game:GetService("Workspace")

local CameraController = {}
local localPlayer = Players.LocalPlayer
local camera = Workspace.CurrentCamera

local ClientState = require(script.Parent.ClientState)

--------------------------------------------------------------------------------
-- 1. УЛЬТРА-ПЛАВНЫЙ ДВИЖОК ПРУЖИН (SPRING DYNAMICS)
--------------------------------------------------------------------------------
local Spring = {}
Spring.__index = Spring

function Spring.new(mass: number?, force: number?, damping: number?, speed: number?)
	return setmetatable({
		Target = Vector3.zero,
		Position = Vector3.zero,
		Velocity = Vector3.zero,
		Mass = mass or 1,
		Force = force or 30,
		Damping = damping or 5,
		Speed = speed or 3.5,
	}, Spring)
end

function Spring:Update(dt: number): Vector3
	dt = math.min(dt, 0.05)
	local d = (self.Target - self.Position)
	local f = d * self.Force
	local damp = self.Velocity * self.Damping
	local a = (f - damp) / self.Mass
	self.Velocity += a * dt * self.Speed
	self.Position += self.Velocity * dt * self.Speed
	return self.Position
end

function Spring:Impulse(imp: Vector3)
	self.Velocity += imp
end

function Spring:Reset()
	self.Target = Vector3.zero
	self.Position = Vector3.zero
	self.Velocity = Vector3.zero
end

--------------------------------------------------------------------------------
-- 2. НАСТРОЙКИ FOV И ФИЗИКИ
--------------------------------------------------------------------------------
local BASE_FOV = 100
local WALK_FOV = 110
local SPRINT_FOV = 120

local BOB_WALK_SPEED = 8.5
local BOB_SPRINT_SPEED = 13.0

local MouseDivisor = 0.18
local Smoothness = 0.045
local CanViewBody = true

local swaySpring = Spring.new(1, 35, 6.0, 4.0)
local walkSpring = Spring.new(1, 28, 5.0, 3.5)
local landSpring = Spring.new(1, 45, 6.5, 4.0)
local leanSpring = Spring.new(1, 25, 4.5, 3.0)

local AngleX = 0
local TargetAngleX = 0
local AngleY = 0
local TargetAngleY = 0

local bobCycle = 0
local previousFallSpeed = 0
local currentFOV = BASE_FOV
local currentRenderConn: RBXScriptConnection? = nil
local currentStateConn: RBXScriptConnection? = nil
local currentDescConn: RBXScriptConnection? = nil

local hidden = {}

--------------------------------------------------------------------------------
-- 3. ЦЕНТРАЛЬНАЯ ТОЧКА И ВИНЬЕТКА
--------------------------------------------------------------------------------
local function CreateHUDOverlay()
	local playerGui = localPlayer:WaitForChild("PlayerGui")
	if playerGui:FindFirstChild("CameraOverlay") then return end

	local screen = Instance.new("ScreenGui")
	screen.Name = "CameraOverlay"
	screen.ResetOnSpawn = false
	screen.IgnoreGuiInset = true
	screen.DisplayOrder = 99
	screen.Parent = playerGui

	local vignette = Instance.new("ImageLabel")
	vignette.Name = "VignetteImage"
	vignette.Size = UDim2.new(1, 0, 1, 0)
	vignette.BackgroundTransparency = 1
	vignette.Image = "rbxassetid://4576475446"
	vignette.ImageColor3 = Color3.fromRGB(5, 5, 12)
	vignette.ImageTransparency = 0.45
	vignette.Parent = screen

	local dot = Instance.new("Frame")
	dot.Name = "CenterDot"
	dot.Size = UDim2.new(0, 4, 0, 4)
	dot.Position = UDim2.new(0.5, -2, 0.5, -2)
	dot.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
	dot.BackgroundTransparency = 0.2
	dot.BorderSizePixel = 0
	dot.Parent = screen

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(1, 0)
	corner.Parent = dot

	local stroke = Instance.new("UIStroke")
	stroke.Color = Color3.fromRGB(0, 0, 0)
	stroke.Thickness = 1
	stroke.Transparency = 0.5
	stroke.Parent = dot
end

--------------------------------------------------------------------------------
-- 4. СКРЫТИЕ ГОЛОВЫ И ОТОБРАЖЕНИЕ ТЕЛА
--------------------------------------------------------------------------------
local function watchPart(d: Instance)
	if d:IsA("BasePart") then
		if d.Name == "Head" or d:FindFirstAncestorWhichIsA("Accoutrement") then
			hidden[d] = true
		end
	elseif (d:IsA("Decal") or d:IsA("Texture")) and d.Parent and d.Parent.Name == "Head" then
		hidden[d] = true
	end
end

local function UpdateCharacterVisibility(char: Model)
	for i in hidden do
		if i.Parent then
			(i :: any).LocalTransparencyModifier = 1
		else
			hidden[i] = nil
		end
	end
	if not CanViewBody then
		for _, v in ipairs(char:GetChildren()) do
			if v:IsA("BasePart") and v.Name ~= "Head" then
				v.LocalTransparencyModifier = 1
			end
		end
	end
end

local function cleanupCamera()
	if currentRenderConn then currentRenderConn:Disconnect(); currentRenderConn = nil end
	if currentStateConn then currentStateConn:Disconnect(); currentStateConn = nil end
	if currentDescConn then currentDescConn:Disconnect(); currentDescConn = nil end
	for i in hidden do
		if i.Parent then (i :: any).LocalTransparencyModifier = 0 end
	end
	table.clear(hidden)
	swaySpring:Reset()
	walkSpring:Reset()
	landSpring:Reset()
	leanSpring:Reset()
	camera = Workspace.CurrentCamera
	if camera then
		camera.CameraType = Enum.CameraType.Custom
		camera.CameraSubject = nil
	end
end

--------------------------------------------------------------------------------
-- 5. ПРИВЯЗКА К ПЕРСОНАЖУ И РЕНДЕР
--------------------------------------------------------------------------------
local function SetupCharacter(char: Model)
	cleanupCamera()

	local root = char:WaitForChild("HumanoidRootPart", 10) :: BasePart?
	local head = char:WaitForChild("Head", 10) :: BasePart?
	local hum = char:WaitForChild("Humanoid", 10) :: Humanoid?

	if not root or not head or not hum then return end

	hum.AutoRotate = false
	camera = Workspace.CurrentCamera
	if camera then
		camera.CameraType = Enum.CameraType.Scriptable
		camera.CameraSubject = hum
		camera.FieldOfView = BASE_FOV
	end

	table.clear(hidden)
	for _, d in char:GetDescendants() do watchPart(d) end
	currentDescConn = char.DescendantAdded:Connect(watchPart)

	if not ClientState.IsUILocked() then
		UserInputService.MouseIconEnabled = false
		UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
	end

	local _, curYaw, _ = root.CFrame:ToOrientation()
	AngleY = math.deg(curYaw)
	TargetAngleY = AngleY
	AngleX = 0
	TargetAngleX = 0
	previousFallSpeed = 0
	bobCycle = 0

	currentStateConn = hum.StateChanged:Connect(function(_, newState)
		if newState == Enum.HumanoidStateType.Landed then
			local impactForce = math.clamp(math.abs(previousFallSpeed) * 0.035, 0.3, 2.2)
			landSpring:Impulse(Vector3.new(-impactForce * 0.5, -impactForce * 0.9, 0))
		elseif newState == Enum.HumanoidStateType.Jumping then
			landSpring:Impulse(Vector3.new(0.35, 0.2, 0))
		elseif newState == Enum.HumanoidStateType.Dead then
			cleanupCamera()
		end
	end)

	currentRenderConn = RunService.RenderStepped:Connect(function(dt)
		if not char.Parent or not root.Parent or hum.Health <= 0 then
			cleanupCamera()
			return
		end

		local uiLocked = ClientState.IsUILocked()
		if not uiLocked then
			UserInputService.MouseIconEnabled = false
			UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
		end

		UpdateCharacterVisibility(char)

		-- 1. Сверхплавное сглаживание углов мыши с корректной дельтой через 360 градусов
		local smoothFactor = math.min(dt * 22, 1)
		AngleX += (TargetAngleX - AngleX) * smoothFactor
		local diffY = (TargetAngleY - AngleY + 180) % 360 - 180
		AngleY = (AngleY + diffY * smoothFactor) % 360

		-- 2. Определение скорости и состояния бега
		local horizontalVelocity = (root.AssemblyLinearVelocity * Vector3.new(1, 0, 1)).Magnitude
		local isMoving = horizontalVelocity > 1.2
		local isSprinting = hum.WalkSpeed > 18
		local bobFreq = if isSprinting then BOB_SPRINT_SPEED else BOB_WALK_SPEED

		-- 3. Плавный шаговый Bobbing
		if isMoving then
			bobCycle += dt * bobFreq
			local ampY = if isSprinting then 0.035 else 0.018
			local ampX = if isSprinting then 0.025 else 0.012
			local ampRoll = if isSprinting then 0.030 else 0.012

			walkSpring.Target = Vector3.new(
				math.sin(bobCycle) * ampY,
				math.cos(bobCycle * 0.5) * ampX,
				math.sin(bobCycle * 0.5) * ampRoll
			)
		else
			walkSpring.Target = Vector3.zero
		end

		-- 4. Мягкий стрейф-крен
		local relativeMove = root.CFrame:VectorToObjectSpace(root.AssemblyLinearVelocity)
		leanSpring.Target = Vector3.new(0, 0, -math.clamp(relativeMove.X * 0.003, -0.045, 0.045))

		-- 5. Инерция шеи при движении мыши (Mouse Sway)
		if not uiLocked then
			local mouseDelta = UserInputService:GetMouseDelta()
			swaySpring.Target = Vector3.new(
				-math.clamp(mouseDelta.Y * 0.002, -0.045, 0.045),
				-math.clamp(mouseDelta.X * 0.002, -0.045, 0.045),
				-math.clamp(mouseDelta.X * 0.0035, -0.06, 0.06)
			)
		else
			swaySpring.Target = Vector3.zero
		end

		previousFallSpeed = root.AssemblyLinearVelocity.Y

		-- 6. Обновление физических пружин
		local currentSway = swaySpring:Update(dt)
		local currentWalk = walkSpring:Update(dt)
		local currentLand = landSpring:Update(dt)
		local currentLean = leanSpring:Update(dt)

		swaySpring.Target = swaySpring.Target:Lerp(Vector3.zero, math.min(dt * 7, 1))
		landSpring.Target = landSpring.Target:Lerp(Vector3.zero, math.min(dt * 8, 1))

		-- 7. Фиксация на голове + физика пружин
		local baseCamCFrame = CFrame.new(head.Position)
			* CFrame.Angles(0, math.rad(AngleY), 0)
			* CFrame.Angles(math.rad(AngleX), 0, 0)
			* CFrame.new(0, 0.35, -0.6)

		local physicsOffset = Vector3.new(currentWalk.Y, currentWalk.X + currentLand.Y, 0)
		local physicsRotation = CFrame.Angles(
			currentSway.X + currentWalk.X + currentLand.X,
			currentSway.Y,
			currentSway.Z + currentWalk.Z + currentLean.Z
		)

		camera.CFrame = (baseCamCFrame * CFrame.new(physicsOffset)) * physicsRotation

		-- Поворот тела за взглядом
		if not uiLocked then
			root.CFrame = CFrame.new(root.Position) * CFrame.Angles(0, math.rad(AngleY), 0)
		end

		-- 8. Плавный динамический FOV (100 -> 110 -> 120)
		local targetFOV = BASE_FOV
		if isMoving then
			targetFOV = if isSprinting then SPRINT_FOV else WALK_FOV
		end

		currentFOV = currentFOV + (targetFOV - currentFOV) * math.min(dt * 5.5, 1)
		camera.FieldOfView = currentFOV
	end)
end

--------------------------------------------------------------------------------
-- 6. ИНИЦИАЛИЗАЦИЯ
--------------------------------------------------------------------------------
function CameraController.Init()
	CreateHUDOverlay()

	UserInputService.InputChanged:Connect(function(inputObject, gameProcessed)
		if gameProcessed or ClientState.IsUILocked() then return end
		if inputObject.UserInputType == Enum.UserInputType.MouseMovement then
			local deltaX = inputObject.Delta.X / MouseDivisor * Smoothness
			local deltaY = inputObject.Delta.Y / MouseDivisor * Smoothness

			local newX = TargetAngleX - deltaY
			TargetAngleX = math.clamp(newX, -80, 80)
			TargetAngleY = (TargetAngleY - deltaX) % 360
		end
	end)

	localPlayer.CharacterAdded:Connect(SetupCharacter)
	localPlayer.CharacterRemoving:Connect(cleanupCamera)
	if localPlayer.Character then
		SetupCharacter(localPlayer.Character)
	end
end

return CameraController
