--!nonstrict
local Players = game:GetService("Players")
local UIS = game:GetService("UserInputService")
local Run = game:GetService("RunService")
local Workspace = game:GetService("Workspace")
local CS = game:GetService("CollectionService")

local Shared = game.ReplicatedStorage.LastWaveShared
local Config = require(Shared.Config)
local Remotes = require(Shared.Remotes)
local Placement = require(Shared.PlacementRules)
local ClientState = require(script.Parent.ClientState)

local M = {}
local B = Config.BARRICADE
local G = B.Ghost
local H = Config.HAMMER
local player = Players.LocalPlayer
local camera = Workspace.CurrentCamera

local carried, carryHL, hammerTarget = nil, nil, nil
local reason
local yaw = 0
local height = 0
local lastSolve = 0
local currentTargetCF: CFrame? = nil

-- Фриз-режим (Lock / Freeze)
local isFrozen = false
local frozenCF: CFrame? = nil
local frozenAim: Vector3? = nil
local freezeOriginPos: Vector3? = nil
local FREEZE_MAX_DIST = 16

local ROT_STEP = 15
local HEIGHT_SPEED = 2
local MAX_HEIGHT = 4

local function char() return player.Character end
local function hammerEquipped() local c = char(); return c and c:GetAttribute("HammerEquipped") == true end
local function head() local c = char(); local p = c and (c:FindFirstChild("Head") or c:FindFirstChild("HumanoidRootPart")); return p and p:IsA("BasePart") and p or nil end

local function aim()
	if isFrozen and frozenAim then return frozenAim end
	local v = camera.ViewportSize
	return camera:ViewportPointToRay(v.X * 0.5, v.Y * 0.5).Direction.Unit
end

local function ray(dist)
	local h = head()
	if not h then return nil end
	local p = RaycastParams.new()
	p.FilterType = Enum.RaycastFilterType.Exclude
	p.FilterDescendantsInstances = { char(), carried }
	local r = Workspace:Raycast(h.Position, aim() * dist, p)
	return r and r.Instance
end

local function tagged(i, tag)
	local c = i
	while c and c ~= Workspace do
		if CS:HasTag(c, tag) then return c end
		c = c.Parent
	end
end

local function nearest(dist, nailed)
	local c = char()
	local r = c and c:FindFirstChild("HumanoidRootPart")
	if not r then return nil end
	local best, bd = nil, dist
	for _, p in CS:GetTagged("BarricadeProp") do
		if p ~= carried and (nailed == nil or (p:GetAttribute("IsNailed") == true) == nailed) then
			local d = Placement.ReachDistance(r.Position, p)
			if d < bd then best, bd = p, d end
		end
	end
	return best
end

local function setHighlight(p: Instance?, valid: boolean)
	if not p then
		if carryHL then carryHL:Destroy(); carryHL = nil end
		return
	end
	if not carryHL or carryHL.Adornee ~= p then
		if carryHL then carryHL:Destroy() end
		local h = Instance.new("Highlight")
		h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		h.FillTransparency = 0.82
		h.OutlineTransparency = 0.1
		h.Adornee = p
		h.Parent = p
		carryHL = h
	end
	local col = if valid then G.ValidColor else G.InvalidColor
	carryHL.FillColor = col
	carryHL.OutlineColor = col
end

local function solve()
	if not carried then return end
	local h = head()
	if not h then return end

	local cf, r = Placement.Solve({
		origin = h.Position,
		dir = aim(),
		maxDist = B.MaxPlaceDistance,
		size = Placement.SizeOf(carried),
		yaw = yaw,
		ignore = { char(), carried }
	})
	reason = r
	if cf then
		local targetCF = cf * CFrame.new(0, height, 0)
		setHighlight(carried, true)
		return targetCF
	else
		setHighlight(carried, false)
		return nil
	end
end

function M.GetHint()
	local c = char()
	if c and c:GetAttribute("Role") == "Zombie" then return "ЛКМ бить · R способность", nil end
	if carried then
		if isFrozen then
			return "🔒 [SHIFT] ЗАФИКСИРОВАНО · ЛКМ/Y поставить · R вращать · SHIFT снять", reason
		else
			return "ЛКМ / Y поставить · [SHIFT] зафиксировать · R поворот · E бросить", reason
		end
	end
	if hammerEquipped() then return "ЛКМ чинить · ПКМ снять шуруп", nil end
	local target = hammerTarget
	if target and target:GetAttribute("IsNailed") == true then
		return "ПКМ снять шуруп (демонтаж) · E взять", nil
	end
	return "E взять предмет · [3] молоток для ремонта", nil
end

function M.IsCarrying() return carried ~= nil end
function M.IsFrozen() return isFrozen end
function M.HammerTarget() return hammerTarget end

function M.ToggleFreeze()
	if not carried then return end
	local c = char()
	local r = c and c:FindFirstChild("HumanoidRootPart")
	if not r then return end

	if not isFrozen then
		local solved = solve()
		if solved then
			isFrozen = true
			frozenCF = solved
			frozenAim = aim()
			freezeOriginPos = r.Position
			currentTargetCF = solved
		end
	else
		isFrozen = false
		frozenCF = nil
		frozenAim = nil
		freezeOriginPos = nil
	end
end

function M.TryPickupOrDrop()
	if ClientState.IsUILocked() then return end
	local c = char()
	local carriedVal = c and c:FindFirstChild("CarriedProp")
	local actualCarried = carried or (carriedVal and carriedVal.Value)
	if actualCarried then
		isFrozen = false
		frozenCF = nil
		Remotes.DropRequest:FireServer()
		return
	end
	local p = tagged(ray(Config.MAX_INTERACT_DISTANCE), "BarricadeProp") or nearest(Config.MAX_INTERACT_DISTANCE, false)
	if p then Remotes.PickupRequest:FireServer(p) end
end

function M.TryPlace()
	if carried and not ClientState.IsUILocked() then
		Remotes.PlaceFreeRequest:FireServer(aim(), yaw)
		isFrozen = false
		frozenCF = nil
	end
end

function M.Rotate(d, isShift)
	local step = ROT_STEP * (d or 1)
	if isShift then step = -step end
	yaw = (yaw + math.rad(step)) % (math.pi * 2)
	lastSolve = 0
	if isFrozen and frozenCF then
		isFrozen = false
		M.ToggleFreeze()
	end
end

function M.AdjustHeight(delta)
	height = math.clamp(height + delta, -MAX_HEIGHT, MAX_HEIGHT)
	lastSolve = 0
	if isFrozen and frozenCF then
		isFrozen = false
		M.ToggleFreeze()
	end
end

function M.HammerHit()
	if hammerEquipped() and hammerTarget then
		Remotes.HammerRepairRequest:FireServer(hammerTarget)
	end
end

function M.HammerDismantle(start)
	if not start then
		Remotes.HammerDismantleRequest:FireServer(nil, false)
		return
	end
	local target = hammerTarget or tagged(ray(H.Reach), "BarricadeProp") or nearest(H.Reach, true)
	if target then
		Remotes.HammerDismantleRequest:FireServer(target, true)
	end
end

function M.TryInteract()
	if ClientState.IsUILocked() then return end
	local p = ray(Config.MAX_INTERACT_DISTANCE)
	if p and p:IsA("BasePart") and p:GetAttribute("InteractType") then
		Remotes.InteractRequest:FireServer(p)
		return
	end
	local nodes = Workspace:FindFirstChild("ResourceNodes")
	if p and nodes and p:IsDescendantOf(nodes) then
		Remotes.GatherRequest:FireServer(p)
	end
end

function M.Attack()
	if not ClientState.IsUILocked() then
		Remotes.ZombieAttackRequest:FireServer(ray(Config.MAX_INTERACT_DISTANCE))
	end
end

function M.Ability()
	if not ClientState.IsUILocked() then
		Remotes.AbilityRequest:FireServer(ray(60))
	end
end

local function onCarriedChanged(p)
	if p == carried then return end
	if carried then setHighlight(nil, false) end
	carried = p
	yaw = 0
	height = 0
	isFrozen = false
	frozenCF = nil
	frozenAim = nil
	freezeOriginPos = nil
	currentTargetCF = nil
	if p then
		setHighlight(p, true)
		currentTargetCF = solve()
	end
end

local function watch(c)
	local ref = c:WaitForChild("CarriedProp", 10)
	if not ref then return end
	ref.Changed:Connect(function()
		onCarriedChanged(ref.Value)
	end)
	if ref.Value then
		onCarriedChanged(ref.Value)
	end
end

function M.Init()
	if player.Character then task.spawn(watch, player.Character) end
	player.CharacterAdded:Connect(function(c)
		carried = nil
		height = 0
		isFrozen = false
		frozenCF = nil
		currentTargetCF = nil
		setHighlight(nil, false)
		task.spawn(watch, c)
	end)

	UIS.InputBegan:Connect(function(i, processed)
		if processed or ClientState.IsUILocked() then return end
		local c = char()
		local role = c and c:GetAttribute("Role")

		if i.KeyCode == Enum.KeyCode.E and role == "Survivor" then
			M.TryPickupOrDrop()
		elseif (i.KeyCode == Enum.KeyCode.LeftShift or i.KeyCode == Enum.KeyCode.RightShift) and role == "Survivor" and carried then
			M.ToggleFreeze()
		elseif i.KeyCode == Enum.KeyCode.Y and role == "Survivor" and carried then
			M.TryPlace()
		elseif i.KeyCode == Enum.KeyCode.R then
			if carried then
				local isShift = UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.RightShift)
				M.Rotate(1, isShift)
			elseif role == "Zombie" then
				M.Ability()
			end
		elseif i.KeyCode == Enum.KeyCode.T then
			if carried then
				M.Rotate(-1, false)
			end
		elseif i.KeyCode == Enum.KeyCode.Q then
			M.TryInteract()
		elseif i.KeyCode == Enum.KeyCode.H then
			Remotes.UseMedkitRequest:FireServer()
		elseif i.UserInputType == Enum.UserInputType.MouseButton1 then
			if role == "Zombie" then
				M.Attack()
			elseif carried then
				M.TryPlace()
			elseif hammerEquipped() then
				M.HammerHit()
			end
		elseif i.UserInputType == Enum.UserInputType.MouseButton2 and not carried then
			M.HammerDismantle(true)
		end
	end)

	UIS.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton2 then
			M.HammerDismantle(false)
		end
	end)

	UIS.InputChanged:Connect(function(i, processed)
		if processed or not carried then return end
		if i.UserInputType == Enum.UserInputType.MouseWheel then
			M.AdjustHeight(i.Position.Z * (1 / HEIGHT_SPEED))
		end
	end)

	Run.RenderStepped:Connect(function(dt)
		local c = char()
		local root = c and c:FindFirstChild("HumanoidRootPart")

		if isFrozen and root and freezeOriginPos then
			if (root.Position - freezeOriginPos).Magnitude > FREEZE_MAX_DIST then
				isFrozen = false
				frozenCF = nil
				frozenAim = nil
				freezeOriginPos = nil
			end
		end

		if carried then
			local now = os.clock()
			if isFrozen and frozenCF then
				currentTargetCF = frozenCF
			else
				if now - lastSolve > 0.033 then
					lastSolve = now
					local solved = solve()
					if solved then
						currentTargetCF = solved
					end
				end
			end

			if currentTargetCF then
				local lerpRate = math.clamp(dt * 20, 0, 1)
				if carried:IsA("Model") then
					local cur = carried:GetPivot()
					carried:PivotTo(cur:Lerp(currentTargetCF, lerpRate))
				elseif carried:IsA("BasePart") then
					carried.CFrame = carried.CFrame:Lerp(currentTargetCF, lerpRate)
				end
			end
		end

		local reachDist = H.Reach or 14
		hammerTarget = tagged(ray(reachDist), "BarricadeProp") or nearest(reachDist, true)
	end)
end

return M
