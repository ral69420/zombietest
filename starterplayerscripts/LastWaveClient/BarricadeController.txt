local Players=game:GetService("Players");local UIS=game:GetService("UserInputService");local Run=game:GetService("RunService");local Workspace=game:GetService("Workspace");local CS=game:GetService("CollectionService")
local Shared=game.ReplicatedStorage.LastWaveShared;local Config=require(Shared.Config);local Remotes=require(Shared.Remotes);local Placement=require(Shared.PlacementRules);local ClientState=require(script.Parent.ClientState)
local M={};local B=Config.BARRICADE;local G=B.Ghost;local H=Config.HAMMER;local player=Players.LocalPlayer;local camera=Workspace.CurrentCamera
local carried,ghost,ghostHL,carryHL,ghostCF,hammerTarget=nil,nil,nil,nil,nil,nil;local reason;local yaw=0;local lastSolve=0;local dismantleHeld=false
local function char()return player.Character end
local function hammerEquipped()local c=char();return c and c:GetAttribute("HammerEquipped")==true end

local function clearGhost()if ghost then ghost:Destroy()end;ghost=nil;ghostHL=nil;ghostCF=nil end

 clearGhost();local ok,g=pcall(function()return p:Clone()end);if not ok or not g then return end;for _,x in g:GetDescendants()do if x:IsA("BasePart")then x.Anchored=true;x.CanCollide=false;x.CanQuery=false;x.CanTouch=false;x.Transparency=G.PreviewTransparency elseif x:IsA("Script")or x:IsA("LocalScript")or x:IsA("Sound")then x:Destroy()end end;local h=Instance.new("Highlight");h.FillTransparency=G.FillTransparency;h.OutlineTransparency=G.OutlineTransparency;h.DepthMode=Enum.HighlightDepthMode.Occluded;h.Adornee=g;h.Parent=g;ghostHL=h;g.Name="BarricadeGhost";g.Parent=Workspace;ghost=g
end

 if carryHL then carryHL:Destroy();carryHL=nil end;if not p then return end;local h=Instance.new("Highlight");h.FillTransparency=0.7;h.OutlineTransparency=0;h.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop;h.Adornee=p;h.Parent=p;carryHL=h

end

function M.Rotate(d,axis)axis=axis or"yaw";if axis=="yaw"then yaw=(yaw+math.rad(B.RotateStep)*d)%(math.pi*2)elseif axis=="pitch"then end;lastSolve=0 end

local function watch(c)
 local ref=c:WaitForChild("CarriedProp",10);if not ref then return end;ref.Changed:Connect(function()local p=ref.Value;if p==carried then return end;if carried then carryVisual(nil)end;carried=p;if p then yaw=0;carryVisual(p);makeGhost(p)else clearGhost()end end);if ref.Value then carried=ref.Value;carryVisual(carried);makeGhost(carried)end
end
function M.Init()
 if player.Character then task.spawn(watch,player.Character)end;player.CharacterAdded:Connect(function(c)carried=nil;clearGhost();carryVisual(nil);task.spawn(watch,c)end)
 UIS.InputBegan:Connect(function(i,processed)if processed or ClientState.IsUILocked()then return end;local c=char();local role=c and c:GetAttribute("Role");if i.KeyCode==Enum.KeyCode.E and role=="Survivor"then M.TryPickupOrDrop()elseif i.KeyCode==Enum.KeyCode.R then if carried then M.Rotate(1,"yaw")elseif role=="Zombie"then M.Ability()end elseif i.KeyCode==Enum.KeyCode.T then if carried then M.Rotate(1,"pitch")end elseif i.KeyCode==Enum.KeyCode.Q then M.TryInteract()elseif i.KeyCode==Enum.KeyCode.H then Remotes.UseMedkitRequest:FireServer()elseif i.UserInputType==Enum.UserInputType.MouseButton1 then if role=="Zombie"then M.Attack()elseif carried then M.TryPlace()elseif hammerEquipped()then M.HammerHit()end elseif i.UserInputType==Enum.UserInputType.MouseButton2 and hammerEquipped()and not carried then M.HammerDismantle(true)end end)

end
return M
